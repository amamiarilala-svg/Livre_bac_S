\section{Primitives — définition et fonctions usuelles}
% ================================================================

\begin{definition}
	Soit $f$ définie sur un intervalle $I$. Une \textbf{primitive} de $f$
	sur $I$ est une fonction $F$ dérivable sur $I$ telle que $F'=f$.
\end{definition}

\begin{theoreme}[Existence et unicité à une constante près]
	Soit $f$ une fonction définie sur $I$.
	\begin{itemize}[label=\textbullet]
		\item Si $f$ est \textbf{continue} sur $I$, alors $f$ admet des
		primitives sur $I$.
		\item Si $F$ est une primitive de $f$ sur $I$, les primitives de $f$ sont
		exactement les fonctions $F+C$, $C\in\R$.
	\end{itemize}
\end{theoreme}

\begin{propriete}{Primitives usuelles}{prop_prim_usuelles}
	\begin{center}
	\renewcommand{\arraystretch}{2.0}
	\setlength{\tabcolsep}{5pt}
	\begin{tabular}{>{\centering\arraybackslash}p{2.9cm}
	                >{\centering\arraybackslash}p{2.6cm}
	                >{\centering\arraybackslash}p{2.7cm}}
	\hline
	$f(x)$ & une primitive $F(x)$ & sur \\
	\hline
	$k$ \ ($k\in\R$)                & $kx$                       & $\R$ \\
	$x^{n}$ \ ($n\in\N$)           & $\dfrac{x^{n+1}}{n+1}$     & $\R$ \\
	$x^{n}$ \ ($n\leq-2$)          & $\dfrac{x^{n+1}}{n+1}$     & $\R^*$ \\
	$x^{\alpha}$ \ ($\alpha\neq-1$) & $\dfrac{x^{\alpha+1}}{\alpha+1}$ & $]0\,;+\infty[$ \\
	$\dfrac{1}{x}$                 & $\ln|x|$                   & $\R^*$ \\
	$\dfrac{1}{\sqrt{x}}$          & $2\sqrt{x}$                & $]0\,;+\infty[$ \\
	$\e^{x}$                       & $\e^{x}$                   & $\R$ \\
	$\cos x$                       & $\sin x$                   & $\R$ \\
	$\sin x$                       & $-\cos x$                  & $\R$ \\
	$\dfrac{1}{\cos^{2}x}$         & $\tan x$                   & $\left]-\tfrac{\pi}{2}\,;\tfrac{\pi}{2}\right[$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{attention}
	Une primitive n'est \textbf{jamais unique} : on écrit toujours \og $+C$ \fg\
	tant qu'aucune condition n'est imposée.
\end{attention}

\section{Opérations. Primitives de la forme \texorpdfstring{$u'\times f(u)$}{u' f(u)}}
% ================================================================

\begin{propriete}{Linéarité}{prop_prim_lin}
	Si $F$, $G$ sont des primitives de $f$, $g$ sur $I$ et $\lambda,\mu\in\R$,
	alors $\lambda F+\mu G$ est une primitive de $\lambda f+\mu g$ sur $I$.
\end{propriete}

\begin{propriete}{Primitives composées (\og inverse des dérivées \fg)}{prop_prim_comp}
	Soit $u$ dérivable sur $I$ :
	\begin{center}
	\renewcommand{\arraystretch}{2.1}
	\setlength{\tabcolsep}{6pt}
	\begin{tabular}{>{\centering\arraybackslash}p{3.4cm}
	                >{\centering\arraybackslash}p{4.0cm}}
	\hline
	$f$ & une primitive $F$ \\
	\hline
	$u'\,u^{n}$ \ ($n\neq-1$)     & $\dfrac{u^{n+1}}{n+1}$ \\
	$\dfrac{u'}{u^{2}}$           & $-\dfrac{1}{u}$ \\
	$\dfrac{u'}{\sqrt{u}}$        & $2\sqrt{u}$ \\
	$\dfrac{u'}{u}$               & $\ln|u|$ \\
	$u'\,\e^{u}$                  & $\e^{u}$ \\
	$u'\cos u$                    & $\sin u$ \\
	$u'\sin u$                    & $-\cos u$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{methode}
	Pour primitiver une expression : repérer une fonction intérieure $u$,
	calculer $u'$, écrire l'expression sous la forme
	$\lambda\times\bigl(u'\times f(u)\bigr)$, puis appliquer le tableau et
	ajuster le coefficient.
\end{methode}

\begin{exemple}
	\[
	\begin{aligned}
	3x^{2}(x^{3}+1)^{4} &\ \longrightarrow\ \frac{(x^{3}+1)^{5}}{5}
	&&(u=x^{3}+1),\\[0.3em]
	\frac{2x}{x^{2}+1} &\ \longrightarrow\ \ln(x^{2}+1)
	&&(u=x^{2}+1),\\[0.3em]
	(2x+1)\,\e^{x^{2}+x} &\ \longrightarrow\ \e^{x^{2}+x}
	&&(u=x^{2}+x),\\[0.3em]
	\frac{x}{\sqrt{x^{2}+3}} &\ \longrightarrow\ \sqrt{x^{2}+3}
	&&\left(\tfrac12\cdot\dfrac{2x}{\sqrt{x^{2}+3}}\right).
	\end{aligned}
	\]
\end{exemple}

\section{Primitive vérifiant une condition}
% ================================================================

\begin{propriete}{Unicité sous condition}{prop_prim_cond}
	Si $f$ est continue sur $I$, $x_0\in I$ et $y_0\in\R$, il existe une
	\textbf{unique} primitive $F$ de $f$ sur $I$ telle que $F(x_0)=y_0$.
\end{propriete}

\begin{exemple}
	Primitive $F$ de $f(x)=\e^{x}-\dfrac{1}{x}$ sur $]0\,;+\infty[$ telle que
	$F(1)=\e$.

	$F(x)=\e^{x}-\ln x+C$ ; \ $F(1)=\e-0+C=\e\Rightarrow C=0$ ;
	donc $F(x)=\e^{x}-\ln x$.
\end{exemple}

\section{Intégrale d'une fonction continue}
% ================================================================

\begin{definition}
	Soit $f$ \textbf{continue} sur $[a\,;b]$ et $F$ une primitive de $f$.
	L'\textbf{intégrale} de $f$ de $a$ à $b$ est le réel
	\[
	\int_a^b f(x)\dd x=\Bigl[F(x)\Bigr]_a^b=F(b)-F(a).
	\]
	Ce résultat ne dépend pas de la primitive choisie.
\end{definition}

\begin{propriete}{Règles de calcul}{prop_int_regles}
	Pour $f$, $g$ continues sur un intervalle contenant $a$, $b$, $c$ et
	$\lambda\in\R$ :
	\begin{itemize}[label=\textbullet]
		\item \textbf{Linéarité :}
		$\displaystyle\int_a^b(\lambda f+g)=\lambda\int_a^b f+\int_a^b g$ ;
		\item \textbf{Chasles :}
		$\displaystyle\int_a^b f=\int_a^c f+\int_c^b f$ ;
		\item \textbf{Bornes :}
		$\displaystyle\int_a^a f=0$, \quad
		$\displaystyle\int_b^a f=-\int_a^b f$ ;
		\item \textbf{Positivité :} si $a\leq b$ et $f\geq 0$ sur $[a\,;b]$, alors
		$\displaystyle\int_a^b f\geq 0$ ;
		\item \textbf{Comparaison :} si $a\leq b$ et $f\leq g$ sur $[a\,;b]$, alors
		$\displaystyle\int_a^b f\leq\int_a^b g$ ;
		\item \textbf{Inégalité de la moyenne :} si $m\leq f\leq M$ sur $[a\,;b]$,
		alors $m(b-a)\leq\displaystyle\int_a^b f\leq M(b-a)$.
	\end{itemize}
\end{propriete}

\begin{exemple}
	\[
	\int_1^{\e}\frac{\dd x}{x}=\bigl[\ln x\bigr]_1^{\e}=1,
	\qquad
	\int_0^1\e^{x}\dd x=\bigl[\e^{x}\bigr]_0^1=\e-1.
	\]
\end{exemple}

\section{Fonction définie par une intégrale}
% ================================================================

\begin{theoreme}[Théorème fondamental]
	Si $f$ est continue sur $I$ et $a\in I$, la fonction
	\[
	\Phi:x\longmapsto\int_a^x f(t)\dd t
	\]
	est \textbf{la} primitive de $f$ sur $I$ qui s'annule en $a$. En
	particulier $\Phi$ est dérivable et $\Phi'(x)=f(x)$.
\end{theoreme}

\begin{propriete}{Bornes variables}{prop_bornes_var}
	Si $u$, $v$ sont dérivables à valeurs dans $I$ et $f$ continue sur $I$,
	alors $\displaystyle G(x)=\int_{u(x)}^{v(x)}f(t)\dd t$ est dérivable et
	\[
	G'(x)=v'(x)\,f\bigl(v(x)\bigr)-u'(x)\,f\bigl(u(x)\bigr).
	\]
\end{propriete}

\begin{exemple}
	\textbf{a)} $\Phi(x)=\displaystyle\int_1^{x}\frac{\ln t}{t}\dd t$ :
	$\Phi'(x)=\dfrac{\ln x}{x}$, et
	$\Phi(x)=\left[\dfrac{(\ln t)^{2}}{2}\right]_1^{x}=\dfrac{(\ln x)^{2}}{2}$.

	\textbf{b)} $G(x)=\displaystyle\int_{x}^{x^{2}}\frac{\dd t}{t}$ ($x>1$) :
	$G'(x)=2x\cdot\dfrac{1}{x^{2}}-1\cdot\dfrac{1}{x}=\dfrac{1}{x}$
	(on retrouve $G(x)=\ln(x^{2})-\ln x=\ln x$).
\end{exemple}

\section{Intégration par parties}
% ================================================================

\begin{theoreme}[Intégration par parties (IPP)]
	Si $u$, $v$ sont dérivables sur $[a\,;b]$, à dérivées continues :
	\[
	\int_a^b u(x)\,v'(x)\dd x
	=\Bigl[u(x)\,v(x)\Bigr]_a^b-\int_a^b u'(x)\,v(x)\dd x.
	\]
\end{theoreme}

\begin{methode}
	Choisir $u$ et $v'$ pour que $u'$ soit \textbf{plus simple} que $u$ et $v$
	\textbf{facile} à obtenir. Cas classiques : $u=$ polynôme ou $u=\ln x$ ;
	$v'=\e^{x}$, $\cos x$, $\sin x$, $x^{\alpha}$.
\end{methode}

\begin{exemple}
	\textbf{a)} $\displaystyle\int_1^{\e}\ln x\dd x$ : $u=\ln x$, $v'=1$,
	$u'=\tfrac1x$, $v=x$.
	\[
	\int_1^{\e}\ln x\dd x=\bigl[x\ln x\bigr]_1^{\e}-\int_1^{\e}1\dd x
	=\e-(\e-1)=1.
	\]

	\textbf{b)} $\displaystyle\int_0^1 x\,\e^{x}\dd x$ : $u=x$, $v'=\e^{x}$.
	\[
	\int_0^1 x\,\e^{x}\dd x=\bigl[x\e^{x}\bigr]_0^1-\int_0^1\e^{x}\dd x
	=\e-(\e-1)=1.
	\]
\end{exemple}

\section{Changement de variable affine}
% ================================================================

\begin{propriete}{Intégrale de $x\mapsto f(\alpha x+\beta)$}{prop_cvaffine}
	Si $F$ est une primitive de $f$ et $\alpha\neq 0$, alors une primitive de
	$x\mapsto f(\alpha x+\beta)$ est $x\mapsto\dfrac{1}{\alpha}F(\alpha x+\beta)$.
	Par suite
	\[
	\int_a^b f(\alpha x+\beta)\dd x
	=\frac{1}{\alpha}\Bigl[F(\alpha x+\beta)\Bigr]_a^b .
	\]
	(Cela revient à poser $t=\alpha x+\beta$, d'où $\mathrm{d}t=\alpha\,\mathrm{d}x$.)
\end{propriete}

\begin{exemple}
	\[
	\int_0^1\e^{2x+1}\dd x=\frac12\bigl[\e^{2x+1}\bigr]_0^1=\frac{\e^{3}-\e}{2},
	\qquad
	\int (3x-1)^{5}\dd x=\frac{(3x-1)^{6}}{18}+C.
	\]
\end{exemple}

\section{Aire, valeur moyenne, volume, moments d'inertie}
% ================================================================

\begin{propriete}{Aire}{prop_aire}
	Si $f$ est continue \textbf{positive} sur $[a\,;b]$, l'aire (en unités
	d'aire) du domaine délimité par $\mathcal C_f$, l'axe $(Ox)$ et les droites
	$x=a$, $x=b$ vaut $\displaystyle\mathcal A=\int_a^b f(x)\dd x$.

	Si $f\geq g$ sur $[a\,;b]$, l'aire entre $\mathcal C_f$ et $\mathcal C_g$
	vaut $\displaystyle\int_a^b\bigl[f(x)-g(x)\bigr]\dd x$.
\end{propriete}

\begin{center}
\begin{tikzpicture}[>=Stealth, scale=1.0]
	\begin{scope}
		\clip (1,-0.1) rectangle (3,2.6);
		\fill[gray!22] (1,0) -- plot[domain=1:3,samples=60]
		  (\x,{0.7*exp(0.35*\x)}) -- (3,0) -- cycle;
	\end{scope}
	\draw[->] (-0.2,0) -- (4.3,0) node[right,font=\small] {$x$};
	\draw[->] (0,-0.2) -- (0,2.9) node[above,font=\small] {$y$};
	\draw[thick,domain=0.2:3.9,samples=90] plot (\x,{0.7*exp(0.35*\x)});
	\draw[dashed] (1,0) node[below,font=\small] {$a$} -- (1,{0.7*exp(0.35)});
	\draw[dashed] (3,0) node[below,font=\small] {$b$} -- (3,{0.7*exp(1.05)});
	\node[font=\small] at (2,0.7) {$\mathcal A$};
	\node[font=\scriptsize] at (3.75,{0.7*exp(0.35*3.75)-0.25}) {$\mathcal C_f$};
\end{tikzpicture}
\end{center}

\begin{attention}
	Si $f$ change de signe sur $[a\,;b]$, l'aire géométrique s'obtient en
	\textbf{découpant} aux zéros de $f$ et en sommant les $\bigl|\int\bigr|$
	partielles.
\end{attention}

\begin{definition}
	La \textbf{valeur moyenne} de $f$ continue sur $[a\,;b]$ ($a<b$) est
	\[
	\mu=\frac{1}{b-a}\int_a^b f(x)\dd x.
	\]
\end{definition}

\begin{propriete}{Volume d'un solide de révolution}{prop_volume}
	Si $f$ est continue positive sur $[a\,;b]$, le volume du solide engendré
	par la rotation de $\mathcal C_f$ autour de $(Ox)$ vaut
	\[
	V=\pi\int_a^b\bigl[f(x)\bigr]^{2}\dd x.
	\]
\end{propriete}

\begin{propriete}{Moments d'inertie d'une plaque}{prop_inertie}
	Pour la plaque homogène (masse surfacique $1$) délimitée par
	$\mathcal C_f$ ($f\geq 0$), l'axe $(Ox)$ et $x=a$, $x=b$ :
	\[
	I_{(Oy)}=\int_a^b x^{2}\,f(x)\dd x,
	\qquad
	I_{(Ox)}=\int_a^b\frac{\bigl[f(x)\bigr]^{3}}{3}\dd x.
	\]
\end{propriete}

\begin{exemple}
	$f(x)=\e^{x}$ sur $[0\,;1]$ :
	\[
	V=\pi\int_0^1\e^{2x}\dd x=\pi\left[\frac{\e^{2x}}{2}\right]_0^1
	=\frac{\pi(\e^{2}-1)}{2}.
	\]
\end{exemple}

\exerciceresolu
\textbf{Calculs variés}

\[
\text{a) } \int_0^1\bigl(3x^{2}+\e^{x}\bigr)\dd x
\qquad
\text{b) } \int_1^{2}\frac{2x}{x^{2}+1}\dd x
\qquad
\text{c) } \int_0^{1}(x+1)\e^{x}\dd x
\]

\solution

\textbf{a)} $\bigl[x^{3}+\e^{x}\bigr]_0^1=(1+\e)-(0+1)=\e$.

\textbf{b)} $\bigl[\ln(x^{2}+1)\bigr]_1^{2}=\ln 5-\ln 2=\ln\dfrac52$.

\textbf{c)} IPP : $u=x+1$, $v'=\e^{x}$, $u'=1$, $v=\e^{x}$ :
\[
\int_0^1(x+1)\e^{x}\dd x=\bigl[(x+1)\e^{x}\bigr]_0^1-\int_0^1\e^{x}\dd x
=2\e-1-(\e-1)=\e.
\]

\bigskip
\exerciceresolu
\textbf{Aire et volume sous $\mathcal C_{\ln}$}

Soit le domaine $D$ délimité par $y=\ln x$, l'axe $(Ox)$ et $x=\e$.
\begin{enumerate}
	\item Calculer l'aire de $D$.
	\item Calculer le volume engendré par la rotation de $D$ autour de $(Ox)$.
\end{enumerate}

\solution

\textbf{1.} $\ln x\geq 0$ sur $[1\,;\e]$, donc
$\mathcal A=\displaystyle\int_1^{\e}\ln x\dd x=1$ (IPP, voir plus haut).

\textbf{2.} $V=\pi\displaystyle\int_1^{\e}(\ln x)^{2}\dd x$. Par IPP
($u=(\ln x)^{2}$, $v'=1$) :
\[
\int_1^{\e}(\ln x)^{2}\dd x
=\bigl[x(\ln x)^{2}\bigr]_1^{\e}-2\int_1^{\e}\ln x\dd x
=\e-2.
\]
Donc $V=\pi(\e-2)$.

\bigskip
\exerciceresolu
\textbf{Fonction définie par une intégrale}

Soit $F(x)=\displaystyle\int_0^{x}t\,\e^{-t}\dd t$ pour $x\geq 0$.
\begin{enumerate}
	\item Justifier que $F$ est dérivable et calculer $F'(x)$ ; en déduire les
	variations de $F$.
	\item Calculer $F(x)$ explicitement, puis $\displaystyle\lim_{x\to+\infty}F(x)$.
\end{enumerate}

\solution

\textbf{1.} $t\mapsto t\e^{-t}$ est continue, donc $F$ est dérivable avec
$F'(x)=x\,\e^{-x}\geq 0$ : $F$ est croissante sur $[0\,;+\infty[$.

\textbf{2.} IPP ($u=t$, $v'=\e^{-t}$, $v=-\e^{-t}$) :
\[
F(x)=\bigl[-t\e^{-t}\bigr]_0^{x}+\int_0^{x}\e^{-t}\dd t
=-x\e^{-x}+\bigl(1-\e^{-x}\bigr)=1-(x+1)\e^{-x}.
\]
Comme $(x+1)\e^{-x}\to 0$, $\displaystyle\lim_{x\to+\infty}F(x)=1$.

\subsection{Échauffement}

\exercice{Vrai ou faux \hfill \facile}
Répondre par \emph{vrai} ou \emph{faux} en justifiant.
\begin{enumerate}
	\item Une fonction continue sur $[a\,;b]$ admet une infinité de primitives.
	\item $\displaystyle\int_a^b f(x)\dd x$ est toujours positive.
	\item Une primitive de $\dfrac{1}{x}$ sur $\R^*$ est $\ln x$.
	\item $\displaystyle\int_2^{2} f(x)\dd x=0$.
	\item $\displaystyle\left(\int_0^{x}\e^{-t^{2}}\dd t\right)'=\e^{-x^{2}}$.
	\item Si $F'=f$ alors $\displaystyle\int_a^b f=F(a)-F(b)$.
	\item La valeur moyenne de $x\mapsto x$ sur $[0\,;2]$ vaut $1$.
\end{enumerate}

	\item $G(x)=\displaystyle\int_1^{x}\frac{\ln t}{t}\dd t$
	\item $G(x)=\displaystyle\int_{x}^{2x}\frac{\dd t}{t}$ \; ($x>0$)
	\item $G(x)=\displaystyle\int_0^{x^{2}}\e^{-t}\dd t$
\end{enumerate}
\end{multicols}

\subsection{Sujets types Bac}

\exercice{Sujet type Bac — suite d'intégrales \hfill \difficile}
Pour $n\in\N$, on pose $I_n=\displaystyle\int_0^{1}x^{n}\,\e^{-x}\dd x$.

\textbf{Partie A}
\begin{enumerate}
	\item Calculer $I_0$.
	\item À l'aide d'une intégration par parties, montrer que pour tout $n$,
	$I_{n+1}=(n+1)\,I_n-\e^{-1}$.
	\item En déduire $I_1$ et $I_2$.
\end{enumerate}

\textbf{Partie B}
\begin{enumerate}
	\setcounter{enumi}{3}
	\item Montrer que $0\leq I_n\leq\dfrac{1}{n+1}$ pour tout $n$.
	\item En déduire $\displaystyle\lim_{n\to+\infty}I_n$.
	\item Montrer que la suite $(I_n)$ est décroissante.
\end{enumerate}

\exercice{Sujet type Bac — fonction définie par une intégrale \hfill \difficile}
Soit $F(x)=\displaystyle\int_1^{x}\frac{\ln t}{t^{2}}\dd t$ pour $x>0$.

\textbf{Partie A}
\begin{enumerate}
	\item Justifier que $F$ est dérivable sur $]0\,;+\infty[$ et calculer
	$F'(x)$.
	\item Dresser le tableau de variations de $F$ (on précisera le signe de
	$F'$).
	\item À l'aide d'une IPP, montrer que
	$F(x)=1-\dfrac{1+\ln x}{x}$.
\end{enumerate}

\textbf{Partie B}
\begin{enumerate}
	\setcounter{enumi}{3}
	\item Déterminer $\displaystyle\lim_{x\to 0^{+}}F(x)$ et
	$\displaystyle\lim_{x\to+\infty}F(x)$.
	\item En déduire que $F(x)\geq 0$ pour tout $x>0$, l'égalité n'ayant lieu
	que pour $x=1$.
	\item Montrer que l'équation $F(x)=\dfrac{1}{2}$ admet exactement deux
	solutions.
\end{enumerate}

\exercice{Sujet type Bac — aire, volume, encadrement \hfill \difficile}
Soit $f(x)=\dfrac{1}{x+1}$ sur $[0\,;1]$, de courbe $\mathcal C$.

\textbf{Partie A}
\begin{enumerate}
	\item Déterminer une primitive de $f$ et calculer
	$A=\displaystyle\int_0^{1}f(x)\dd x$.
	\item Calculer le volume $V$ engendré par la rotation de $\mathcal C$ autour
	de $(Ox)$.
	\item Calculer la valeur moyenne de $f$ sur $[0\,;1]$.
\end{enumerate}

\textbf{Partie B}
\begin{enumerate}
	\setcounter{enumi}{3}
	\item Montrer que pour tout $x\in[0\,;1]$,
	$1-x\leq\dfrac{1}{x+1}\leq 1$.
	\item En intégrant, en déduire un encadrement de $\ln 2$.
\end{enumerate}

\exercice{Sujet type Bac — somme de Riemann \hfill \difficile}
Pour $n\geq 1$, on pose
$S_n=\dfrac{1}{n}\displaystyle\sum_{k=1}^{n}\e^{k/n}$.
\begin{enumerate}
	\item Interpréter $S_n$ comme une somme d'aires de rectangles sous
	$\mathcal C_{\exp}$ sur $[0\,;1]$.
	\item Admettant que $\displaystyle\lim_{n\to+\infty}S_n=\int_0^{1}\e^{x}\dd x$,
	déterminer cette limite.
	\item De même, déterminer $\displaystyle\lim_{n\to+\infty}
	\sum_{k=1}^{n}\frac{1}{n+k}$.
\end{enumerate}

\exercice{Sujet type Bac — IPP itérée et récurrence \hfill \difficile}
Pour $n\in\N$, on pose $J_n=\displaystyle\int_0^{1}(1-x)^{n}\,\e^{x}\dd x$.
\begin{enumerate}
	\item Calculer $J_0$.
	\item À l'aide d'une IPP, montrer que $J_{n+1}=(n+1)\,J_n-1$ pour tout
	$n$.
	\item Montrer par récurrence que
	$J_n=n!\left(\e-\displaystyle\sum_{k=0}^{n}\frac{1}{k!}\right)$.
	\item Montrer que $0\leq J_n\leq\dfrac{\e}{n+1}$, puis en déduire que
	$\displaystyle\sum_{k=0}^{n}\frac{1}{k!}\longrightarrow\e$.
\end{enumerate}
