\begin{objectifs}
	
	À l'issue de ce chapitre, vous serez capable de :
	
	\begin{itemize}
		\item Déterminer une limite.
		\item Lever une forme indéterminée.
		\item Étudier une asymptote.
		\item Exploiter les limites pour étudier une fonction.
	\end{itemize}
	
\end{objectifs}


Par contre; 
\begin{attention}
	
	La forme
	
	\[
	\frac{\infty}{\infty}
	\]
	
	n'est pas une limite.
	
	Il s'agit d'une forme indéterminée qui nécessite une étude complémentaire.
	
\end{attention}

\lipsum[6-10]
A retenir :
\begin{astucebac}
	
	Avant de calculer une limite de suite,
	essaie d'abord de déterminer si la suite est :
	
	\begin{itemize}
		\item croissante ;
		\item décroissante ;
		\item bornée.
	\end{itemize}
	
	Le théorème de convergence permet souvent
	de conclure très rapidement.
	
\end{astucebac}
