\chapter{Fonction logarithme népérien}

\begin{citationchapitre}
	« Les logarithmes, en abrégeant les calculs, ont pour ainsi dire doublé

	— Pierre-Simon de Laplace
\end{citationchapitre}

	\begin{itemize}[label=\textbullet]

% ================================================================
\section{Définition et sens de variation}
% ================================================================

\begin{definition}
	La fonction \textbf{logarithme népérien}, notée $\ln$, est l'unique
	primitive sur $]0\,;+\infty[$ de $x\mapsto\dfrac{1}{x}$ qui s'annule en $1$ :
	\[
	\forall x>0,\quad (\ln x)'=\frac{1}{x}\qquad\text{et}\qquad \ln 1=0.
	\]
	Le réel $\e$ est défini par $\ln \e = 1$ ($\e\approx 2{,}718$).
\end{definition}

\begin{propriete}{Variation et signe}{prop_ln_signe}
	$\ln$ est \textbf{strictement croissante} sur $]0\,;+\infty[$
	(car $(\ln x)'=\tfrac1x>0$). De plus :
	\[
	\begin{aligned}
	\ln x<0 &\iff 0<x<1,\\
	\ln x=0 &\iff x=1,\\
	\ln x>0 &\iff x>1.
	\end{aligned}
	\]
	\begin{center}
	\renewcommand{\arraystretch}{1.5}
	\setlength{\tabcolsep}{5pt}
	\begin{tabular}{c|ccc}
	$x$        & $0$       &            & $+\infty$ \\
	\hline
	$(\ln x)'$ &           & $+$        &           \\
	\hline
	$\ln x$    & $-\infty$ & $\nearrow$ & $+\infty$ \\
	\end{tabular}
	\end{center}
\end{propriete}

\begin{center}
\begin{tikzpicture}[>=Stealth, scale=1.0]
	\draw[->] (-0.4,0) -- (5.4,0) node[right,font=\small] {$x$};

\end{tikzpicture}
\end{center}

\section{Propriétés algébriques}
% ================================================================

\begin{propriete}{Règles de calcul}{prop_ln_algebre}
	Pour tous $a>0$, $b>0$ et tout $\alpha\in\R$ :
	\begin{center}
	\renewcommand{\arraystretch}{1.9}
	\begin{tabular}{>{\raggedright\arraybackslash}p{3.2cm}
	                >{\centering\arraybackslash}p{4.6cm}}
	\hline
	Produit          & $\ln(ab)=\ln a+\ln b$ \\
	Quotient         & $\ln\!\dfrac{a}{b}=\ln a-\ln b$ \\
	Inverse          & $\ln\!\dfrac{1}{b}=-\ln b$ \\
	Puissance        & $\ln\!\bigl(a^{\alpha}\bigr)=\alpha\ln a$ \\
	Racine carrée    & $\ln\sqrt{a}=\dfrac12\ln a$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{attention}
	$\ln(a+b)\neq\ln a+\ln b$ \quad et \quad $\ln(ab)\neq\ln a\times\ln b$.
	Erreurs très fréquentes au Bac.
\end{attention}

\begin{exemple}
	$A=\ln 6+\ln 3-\ln 2=\ln\dfrac{6\times 3}{2}=\ln 9=2\ln 3$ ;
	\qquad
	$B=2\ln 3-\ln 9=\ln 9-\ln 9=0$.
\end{exemple}

\section{Limites usuelles}
% ================================================================

\begin{propriete}{Limites à connaître}{prop_ln_limites}
	\begin{center}
	\renewcommand{\arraystretch}{2.0}
	\begin{tabular}{>{\raggedright\arraybackslash}p{5.4cm}
	                >{\centering\arraybackslash}p{2.0cm}}
	\hline
	$\displaystyle\lim_{x\to 0^+}\ln x$                       & $-\infty$ \\
	$\displaystyle\lim_{x\to +\infty}\ln x$                   & $+\infty$ \\
	$\displaystyle\lim_{x\to+\infty}\dfrac{\ln x}{x^{n}}$ \ ($n\geq 1$) & $0$ \\
	$\displaystyle\lim_{x\to 0^+}x^{n}\ln x$ \ ($n\geq 1$)    & $0$ \\
	$\displaystyle\lim_{x\to 0}\dfrac{\ln(1+x)}{x}$           & $1$ \\
	$\displaystyle\lim_{x\to 1}\dfrac{\ln x}{x-1}$            & $1$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{remarque}
	Les deux dernières limites sont des \textbf{nombres dérivés}
	($(\ln(1+x))'_{x=0}=1$ et $(\ln x)'_{x=1}=1$). L'usage de ces limites pour
	\textbf{lever une forme indéterminée}, ainsi que les méthodes de
	croissances comparées, sont détaillés au chapitre
	\og Limites et continuité \fg.
\end{remarque}

\section{Dérivée de \texorpdfstring{$\ln u$}{ln u}}
% ================================================================

\begin{theoreme}[Dérivée du logarithme]
	$\ln$ est dérivable sur $]0\,;+\infty[$ et $(\ln x)'=\dfrac{1}{x}$.
\end{theoreme}

\begin{propriete}{Dérivée d'une composée avec $\ln$}{prop_ln_derivee}
	Si $u$ est dérivable :
	\begin{center}
	\renewcommand{\arraystretch}{2.0}
	\begin{tabular}{>{\centering\arraybackslash}p{2.8cm}
	                >{\centering\arraybackslash}p{2.6cm}
	                >{\centering\arraybackslash}p{2.6cm}}
	\hline
	Fonction & Dérivée & Condition \\
	\hline
	$\ln u$        & $\dfrac{u'}{u}$        & $u>0$ \\
	$\ln|u|$       & $\dfrac{u'}{u}$        & $u\neq 0$ \\
	$\ln(ax+b)$    & $\dfrac{a}{ax+b}$      & $ax+b>0$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{methode}
	Pour dériver $x\mapsto\ln\bigl(u(x)\bigr)$ : identifier $u$, calculer $u'$,
	écrire $\dfrac{u'}{u}$.
\end{methode}

\begin{exemple}
	\[
	\bigl(\ln(3x^{2}+1)\bigr)'=\frac{6x}{3x^{2}+1},
	\qquad
	\bigl(x\ln x\bigr)'=\ln x+1,
	\]
	\[
	\left(\ln\frac{x+1}{x-1}\right)'
	=\bigl(\ln(x+1)-\ln(x-1)\bigr)'
	=\frac{1}{x+1}-\frac{1}{x-1}=\frac{-2}{x^{2}-1}.
	\]
\end{exemple}

\begin{remarque}
	Réciproquement, une fonction de la forme $\dfrac{u'}{u}$ admet
	$\ln|u|$ pour primitive : ce point est développé au chapitre
	\og Primitives et intégrale \fg.
\end{remarque}

\section{Équations, inéquations et systèmes}
% ================================================================

\begin{methode}
	Après avoir posé les \textbf{conditions d'existence} ($u(x)>0$\ldots) :
	\begin{itemize}[label=\textbullet]
		\item $\ln\bigl(u(x)\bigr)=a \iff u(x)=\e^{a}$ ;
		\item $\ln\bigl(u(x)\bigr)=\ln\bigl(v(x)\bigr) \iff u(x)=v(x)$ ;
		\item $\ln\bigl(u(x)\bigr)\leq\ln\bigl(v(x)\bigr) \iff u(x)\leq v(x)$
		($\ln$ croissante) ;
		\item pour un \textbf{système} $\{a\ln x+b\ln y=c\}$, poser
		$X=\ln x$, $Y=\ln y$ et résoudre le système linéaire.
	\end{itemize}
\end{methode}

\begin{exemple}
	\textbf{a)} $\ln(2x-1)=\ln(x+3)$ : condition $x>\tfrac12$ ;
	$2x-1=x+3\Rightarrow x=4$. \; $S=\{4\}$.

	\textbf{b)} $\ln x+\ln(x-2)=\ln 3$ : condition $x>2$ ;
	$\ln\bigl[x(x-2)\bigr]=\ln 3\Rightarrow x^{2}-2x-3=0\Rightarrow x=3$
	ou $x=-1$. Avec $x>2$ : $S=\{3\}$.

	\textbf{c)} Système
	$\begin{cases}\ln x+\ln y=\ln 12\\ \ln x-\ln y=\ln 3\end{cases}$ :
	on pose $X=\ln x$, $Y=\ln y$. Alors $X+Y=\ln 12$, $X-Y=\ln 3$, d'où
	$X=\ln 6$, $Y=\ln 2$, soit $x=6$, $y=2$.
\end{exemple}

\section{Logarithme de base \texorpdfstring{$a$}{a} et logarithme décimal}
% ================================================================

\begin{definition}
	Pour $a>0$ et $a\neq 1$, le \textbf{logarithme de base $a$} est
	\[
	\log_a x=\frac{\ln x}{\ln a}\qquad (x>0).
	\]
	Le \textbf{logarithme décimal} est $\log=\log_{10}$.
\end{definition}

\begin{propriete}{Propriétés}{prop_log_base}
	$\log_a$ a les mêmes règles algébriques que $\ln$ ; de plus
	$\log_a a=1$, $\log_a 1=0$ et
	\[
	a^{x}=b \iff x=\log_a b=\frac{\ln b}{\ln a}\qquad (b>0).
	\]
	En particulier $\log 10^{n}=n$.
\end{propriete}

\begin{exemple}
	Résoudre $2^{x}=10$ : $x=\log_2 10=\dfrac{\ln 10}{\ln 2}\approx 3{,}32$.
	\qquad
	$\log 1000=\log 10^{3}=3$.
\end{exemple}

\section{Exercices d'entraînement}
% ================================================================

\subsection{Échauffement}

\exercice{Vrai ou faux \hfill \facile}
Répondre par \emph{vrai} ou \emph{faux} en justifiant.
\begin{enumerate}
	\item $\ln x$ est défini pour tout réel $x$.
	\item $\ln(a b)=\ln a+\ln b$ pour tous réels $a$, $b$.
	\item $\ln x=1\iff x=\e$.
	\item $\displaystyle\lim_{x\to 0^+}\ln x=0$.
	\item $(\ln(3x))'=\dfrac{1}{3x}$.
	\item $\ln x<0$ si et seulement si $0<x<1$.
	\item $\log 100=2$.
\end{enumerate}

\subsection{Calculs directs}

\exercice{Simplifier \hfill \facile}
Écrire sous la forme d'un seul logarithme ou d'un nombre simple.
\begin{multicols}{2}
\begin{enumerate}
	\item $\ln 8+\ln 2-\ln 4$
	\item $\ln 12-\ln 3$
	\item $2\ln 5-\ln 25$
	\item $\ln\sqrt{\e}+\ln\sqrt{\e}$
	\item $\ln(\e^{2})-\ln\!\dfrac{1}{\e}$
	\item $\ln 6+\ln 4-\ln 3-\ln 8$
	\item $\dfrac12\ln 9+\ln\!\dfrac13$
	\item $\ln(\e^{3})+\ln(\e^{-1})$
\end{enumerate}
\end{multicols}

\exercice{Résoudre les équations \hfill \facile}
Dans $\R$, après conditions d'existence.
\begin{multicols}{2}
\begin{enumerate}
	\item $\ln(x-1)=0$
	\item $\ln(2x+1)=\ln 5$
	\item $\ln x+\ln 3=\ln 12$
	\item $\ln(x-2)+\ln(x+1)=\ln 4$
	\item $\ln(x^{2})=\ln(3x+4)$
	\item $(\ln x)^{2}-\ln x-2=0$
	\item $\ln(x+3)-\ln(x-1)=\ln 2$
	\item $2\ln x=\ln(x+6)$
\end{enumerate}
\end{multicols}

\exercice{Résoudre les inéquations \hfill \moyen}
\begin{multicols}{2}
\begin{enumerate}
	\item $\ln x>0$
	\item $\ln(x-1)\leq 0$
	\item $\ln(2x-3)<\ln(x+1)$
	\item $\ln x\geq 1$
	\item $\ln(3-x)>\ln(x+1)$
	\item $(\ln x)^{2}\leq 1$
\end{enumerate}
\end{multicols}

\exercice{Dériver \hfill \moyen}
Préciser l'ensemble de dérivabilité.
\begin{multicols}{2}
\begin{enumerate}
	\item $f(x)=\ln(2x+1)$
	\item $f(x)=\ln(x^{2}+1)$
	\item $f(x)=x\ln x-x$
	\item $f(x)=(\ln x)^{2}$
	\item $f(x)=\dfrac{\ln x}{x}$
	\item $f(x)=\ln\!\dfrac{x-1}{x+1}$ \; ($x>1$)
	\item $f(x)=x^{2}\ln x$
	\item $f(x)=\ln(\ln x)$ \; ($x>1$)
\end{enumerate}
\end{multicols}

\exercice{Limites directes \hfill \moyen}
\begin{multicols}{2}
\begin{enumerate}
	\item $\displaystyle\lim_{x\to 0^+}\bigl(\ln x+1\bigr)$
	\item $\displaystyle\lim_{x\to+\infty}\dfrac{\ln x}{x}$
	\item $\displaystyle\lim_{x\to 0^+}x\ln x$
	\item $\displaystyle\lim_{x\to 1}\dfrac{\ln x}{x-1}$
\end{enumerate}
\end{multicols}
\emph{(Formes indéterminées plus élaborées : voir chapitre
\og Limites et continuité \fg.)}

\subsection{Synthèse}

\exercice{Système \hfill \moyen}
Résoudre dans $]0\,;+\infty[^{2}$ :
\[
\begin{cases}
	2\ln x+\ln y=3\ln 2,\\[0.3em]
	\ln x-\ln y=0.
\end{cases}
\]

\exercice{Une inégalité fondamentale \hfill \moyen}
Soit $f(x)=x-1-\ln x$ sur $]0\,;+\infty[$.
\begin{enumerate}
	\item Calculer $f'(x)$ et dresser le tableau de variations.
	\item En déduire que $\ln x\leq x-1$ pour tout $x>0$, avec égalité si et
	seulement si $x=1$.
\end{enumerate}

\exercice{Logarithme décimal \hfill \moyen}
\begin{enumerate}
	\item Résoudre $10^{x}=2$, puis $10^{2x-1}=50$.
	\item Le nombre $N=2^{40}$ possède-t-il $12$ ou $13$ chiffres en écriture
	décimale ? (On donne $\log 2\approx 0{,}301$.)
\end{enumerate}
