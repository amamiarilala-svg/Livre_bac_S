\chapter{Fonction exponentielle}

\begin{citationchapitre}
	« La fonction exponentielle est la seule, à un coefficient près, qui soit
	égale à sa propre dérivée : elle se reproduit elle-même. »\\[0.4em]
	— adage d'analyse
\end{citationchapitre}

\begin{objectifs}
	\begin{itemize}[label=\textbullet]
		\item Connaître $\exp$ comme \textbf{réciproque} de $\ln$ et ses
		\textbf{propriétés algébriques}.
		\item Connaître les \textbf{limites usuelles} et \textbf{dériver} $\e^{u}$.
		\item Résoudre \textbf{équations, inéquations et systèmes} avec $\e^{x}$.
		\item Connaître les fonctions $x\mapsto a^{x}$ et $x\mapsto u(x)^{v(x)}$.
	\end{itemize}
\end{objectifs}


% ================================================================
\section{Définition}
% ================================================================

\begin{definition}
	La fonction \textbf{exponentielle}, notée $\exp$ ou $x\mapsto\e^{x}$, est la
	bijection \textbf{réciproque} de $\ln$ :
	\[
	\forall x\in\R,\ \forall y>0:\qquad
	y=\e^{x}\iff x=\ln y.
	\]
	Ainsi $\e^{0}=1$, $\e^{1}=\e$, et
	\[
	\ln(\e^{x})=x \ \ (x\in\R),\qquad
	\e^{\ln x}=x \ \ (x>0).
	\]
\end{definition}

\begin{attention}
	$\e^{x}>0$ pour tout $x\in\R$ : l'exponentielle ne s'annule jamais.
\end{attention}

\begin{propriete}{Variation}{prop_exp_var}
	$\exp$ est dérivable sur $\R$, $(\e^{x})'=\e^{x}>0$ : elle est
	\textbf{strictement croissante} sur $\R$, de $0$ (en $-\infty$) à $+\infty$.
	Sa courbe est la symétrique de $\mathcal C_{\ln}$ par rapport à la droite
	$y=x$.
\end{propriete}

\section{Propriétés algébriques}
% ================================================================

\begin{propriete}{Règles de calcul}{prop_exp_algebre}
	Pour tous réels $a$, $b$ et tout $\alpha\in\R$ :
	\begin{center}
	\renewcommand{\arraystretch}{1.9}
	\begin{tabular}{>{\raggedright\arraybackslash}p{3.2cm}
	                >{\centering\arraybackslash}p{4.6cm}}
	\hline
	Produit     & $\e^{a+b}=\e^{a}\,\e^{b}$ \\
	Quotient    & $\e^{a-b}=\dfrac{\e^{a}}{\e^{b}}$ \\
	Inverse     & $\e^{-a}=\dfrac{1}{\e^{a}}$ \\
	Puissance   & $\bigl(\e^{a}\bigr)^{\alpha}=\e^{\alpha a}$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{attention}
	$\e^{a+b}\neq\e^{a}+\e^{b}$ \quad et \quad $\e^{ab}\neq\e^{a}\times\e^{b}$.
\end{attention}

\begin{exemple}
	$A=\e^{3}\,\e^{-1}=\e^{2}$ ;
	\quad
	$B=\dfrac{\e^{2x+1}}{\e^{x-1}}=\e^{x+2}$ ;
	\quad
	$C=\bigl(\e^{3x}\bigr)^{2}\,\e^{-4x}=\e^{6x-4x}=\e^{2x}$.
\end{exemple}

\section{Limites usuelles}
% ================================================================

\begin{propriete}{Limites à connaître}{prop_exp_limites}
	\begin{center}
	\renewcommand{\arraystretch}{2.0}
	\begin{tabular}{>{\raggedright\arraybackslash}p{5.4cm}
	                >{\centering\arraybackslash}p{2.0cm}}
	\hline
	$\displaystyle\lim_{x\to-\infty}\e^{x}$                    & $0$ \\
	$\displaystyle\lim_{x\to+\infty}\e^{x}$                    & $+\infty$ \\
	$\displaystyle\lim_{x\to+\infty}\dfrac{\e^{x}}{x^{n}}$ \ ($n\geq 1$) & $+\infty$ \\
	$\displaystyle\lim_{x\to-\infty}x^{n}\,\e^{x}$ \ ($n\geq 1$) & $0$ \\
	$\displaystyle\lim_{x\to 0}\dfrac{\e^{x}-1}{x}$            & $1$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{remarque}
	$\displaystyle\lim_{x\to 0}\frac{\e^{x}-1}{x}=1$ est le nombre dérivé
	$(\e^{x})'_{x=0}$. La hiérarchie \og l'exponentielle l'emporte sur toute
	puissance \fg\ et l'usage de ces limites pour lever les formes
	indéterminées sont traités au chapitre \og Limites et continuité \fg.
\end{remarque}

\section{Dérivée de \texorpdfstring{$\e^{u}$}{e^u}}
% ================================================================

\begin{theoreme}[Dérivée de l'exponentielle]
	$\exp$ est dérivable sur $\R$ et $(\e^{x})'=\e^{x}$.
\end{theoreme}

\begin{propriete}{Dérivée d'une composée avec $\exp$}{prop_exp_derivee}
	Si $u$ est dérivable sur $\R$ :
	\begin{center}
	\renewcommand{\arraystretch}{2.0}
	\begin{tabular}{>{\centering\arraybackslash}p{3.0cm}
	                >{\centering\arraybackslash}p{3.0cm}}
	\hline
	Fonction & Dérivée \\
	\hline
	$\e^{u}$        & $u'\,\e^{u}$ \\
	$\e^{ax+b}$     & $a\,\e^{ax+b}$ \\
	\hline
	\end{tabular}
	\end{center}
\end{propriete}

\begin{methode}
	Pour dériver $x\mapsto\e^{u(x)}$ : identifier $u$, calculer $u'$, écrire
	$u'\,\e^{u}$.
\end{methode}

\begin{exemple}
	\[
	\bigl(\e^{3x^{2}-1}\bigr)'=6x\,\e^{3x^{2}-1},
	\qquad
	\bigl(x^{2}\e^{-x}\bigr)'=(2x-x^{2})\e^{-x}=x(2-x)\e^{-x},
	\]
	\[
	\left(\frac{\e^{x}-1}{\e^{x}+1}\right)'
	=\frac{\e^{x}(\e^{x}+1)-\e^{x}(\e^{x}-1)}{(\e^{x}+1)^{2}}
	=\frac{2\e^{x}}{(\e^{x}+1)^{2}}.
	\]
\end{exemple}

\begin{remarque}
	Réciproquement, $u'\,\e^{u}$ admet $\e^{u}$ pour primitive : voir le
	chapitre \og Primitives et intégrale \fg. Les équations différentielles
	$y'=ky$ et $y'+ay=b$ sont traitées au chapitre \og Équations
	différentielles \fg.
\end{remarque}

\section{Équations, inéquations et systèmes}
% ================================================================

\begin{methode}
	\begin{itemize}[label=\textbullet]
		\item $\e^{u(x)}=a$ \ ($a>0$) $\iff u(x)=\ln a$ ; \ si $a\leq 0$ : pas de
		solution ;
		\item $\e^{u(x)}=\e^{v(x)} \iff u(x)=v(x)$ ;
		\item $\e^{u(x)}\leq\e^{v(x)} \iff u(x)\leq v(x)$ ($\exp$ croissante) ;
		\item équation en $\e^{2x}$, $\e^{x}$ : poser $X=\e^{x}$ (\textbf{avec}
		$X>0$) ;
		\item système $\{a\,\e^{x}+b\,\e^{y}=c\}$ : poser $X=\e^{x}$, $Y=\e^{y}$
		($X,Y>0$).
	\end{itemize}
\end{methode}

\begin{exemple}
	\textbf{a)} $\e^{2x}-3\e^{x}+2=0$ : on pose $X=\e^{x}>0$,
	$X^{2}-3X+2=0\Rightarrow X=1$ ou $X=2$, d'où $x=0$ ou $x=\ln 2$.

	\textbf{b)} $\e^{x^{2}}\leq\e^{2x}\iff x^{2}\leq 2x\iff x(x-2)\leq 0
	\iff x\in[0\,;2]$.

	\textbf{c)} Système
	$\begin{cases}\e^{x}+\e^{y}=5\\ \e^{x}\,\e^{y}=6\end{cases}$ :
	$X=\e^{x}$, $Y=\e^{y}$ sont les solutions de $t^{2}-5t+6=0$, soit
	$\{2\,;3\}$. D'où $(x\,;y)=(\ln 2\,;\ln 3)$ ou $(\ln 3\,;\ln 2)$.
\end{exemple}


% ================================================================
\section{Fonctions \texorpdfstring{$x\mapsto a^{x}$}{a^x} et \texorpdfstring{$x\mapsto u^{v}$}{u^v}}
% ================================================================

\begin{definition}
	Pour $a>0$ et pour $u(x)>0$ :
	\[
	a^{x}=\e^{x\ln a},
	\qquad
	u(x)^{v(x)}=\e^{\,v(x)\ln u(x)}.
	\]
\end{definition}

\begin{propriete}{Dérivée de $a^{x}$}{prop_puissance}
	\[
	\bigl(a^{x}\bigr)'=\ln a\cdot a^{x},
	\qquad
	\bigl(a^{u}\bigr)'=u'\ln a\cdot a^{u}.
	\]
\end{propriete}

\begin{exemple}
	$\bigl(2^{x}\bigr)'=\ln 2\cdot 2^{x}$.
	\quad
	$\displaystyle\lim_{x\to 0^+}x^{x}=\lim_{x\to 0^+}\e^{x\ln x}=\e^{0}=1$
	(car $x\ln x\to 0$).
\end{exemple}

\section{Exercices d'entraînement}
% ================================================================

\subsection{Échauffement}

\exercice{Vrai ou faux \hfill \facile}
Répondre par \emph{vrai} ou \emph{faux} en justifiant.
\begin{enumerate}
	\item $\e^{x}$ peut être négatif pour $x<0$.
	\item $\e^{a+b}=\e^{a}\,\e^{b}$ pour tous réels $a$, $b$.
	\item $\e^{x}=2\iff x=\ln 2$.
	\item $\displaystyle\lim_{x\to-\infty}\e^{x}=-\infty$.
	\item $(\e^{-x})'=\e^{-x}$.
	\item $\e^{\ln 5}=5$ et $\ln(\e^{5})=5$.
	\item $2^{x}=\e^{x\ln 2}$.
\end{enumerate}

\subsection{Calculs directs}

\exercice{Simplifier \hfill \facile}
\begin{multicols}{2}
\begin{enumerate}
	\item $\e^{2}\,\e^{-3}$
	\item $\dfrac{\e^{5x}}{\e^{2x}}$
	\item $\bigl(\e^{x}\bigr)^{3}\,\e^{-2x}$
	\item $\e^{\ln 4}$
	\item $\e^{2\ln 3}$
	\item $\bigl(\e^{-x}\bigr)^{2}\,\e^{3x}$
	\item $\dfrac{\e^{x+1}}{\e}$
	\item $\e^{\ln x+1}$ \; ($x>0$)
\end{enumerate}
\end{multicols}

\exercice{Résoudre les équations \hfill \facile}
\begin{multicols}{2}
\begin{enumerate}
	\item $\e^{x}=1$
	\item $\e^{2x-1}=\e^{x+2}$
	\item $\e^{x}=5$
	\item $\e^{2x}-4\e^{x}+3=0$
	\item $\e^{2x}+\e^{x}-6=0$
	\item $(\e^{x}-1)(\e^{x}-\e)=0$
	\item $\e^{x^{2}}=\e^{3x-2}$
	\item $\e^{2x}=2\,\e^{x}$
\end{enumerate}
\end{multicols}

\exercice{Résoudre les inéquations \hfill \moyen}
\begin{multicols}{2}
\begin{enumerate}
	\item $\e^{x}<1$
	\item $\e^{x-2}\geq 1$
	\item $\e^{2x}\leq\e^{x+1}$
	\item $\e^{x}>3$
	\item $\e^{x^{2}-1}\leq 1$
	\item $\e^{2x}-3\e^{x}+2\leq 0$
\end{enumerate}
\end{multicols}

\exercice{Dériver \hfill \moyen}
Préciser l'ensemble de dérivabilité.
\begin{multicols}{2}
\begin{enumerate}
	\item $f(x)=\e^{3x}$
	\item $f(x)=\e^{-x^{2}}$
	\item $f(x)=x\,\e^{x}$
	\item $f(x)=\dfrac{\e^{x}}{x}$
	\item $f(x)=(x^{2}-1)\e^{x}$
	\item $f(x)=\dfrac{\e^{x}}{\e^{x}+1}$
	\item $f(x)=\e^{\sqrt{x}}$ \; ($x>0$)
	\item $f(x)=\dfrac{\e^{x}+\e^{-x}}{2}$
\end{enumerate}
\end{multicols}

\exercice{Limites directes \hfill \moyen}
\begin{multicols}{2}
\begin{enumerate}
	\item $\displaystyle\lim_{x\to-\infty}\e^{x}$
	\item $\displaystyle\lim_{x\to+\infty}x\,\e^{-x}$
	\item $\displaystyle\lim_{x\to-\infty}x^{2}\,\e^{x}$
	\item $\displaystyle\lim_{x\to 0}\dfrac{\e^{x}-1}{x}$
\end{enumerate}
\end{multicols}
\emph{(Formes indéterminées plus élaborées : voir chapitre
\og Limites et continuité \fg.)}

\subsection{Synthèse}

\exercice{Système \hfill \moyen}
Résoudre dans $\R^{2}$ :
\[
\begin{cases}
	\e^{x}+\e^{y}=\e^{2}+1,\\[0.3em]
	\e^{x}-\e^{y}=\e^{2}-1.
\end{cases}
\]

\exercice{Une inégalité fondamentale \hfill \moyen}
Soit $g(x)=\e^{x}-x-1$ sur $\R$.
\begin{enumerate}
	\item Calculer $g'(x)$ et dresser le tableau de variations.
	\item En déduire que $\e^{x}\geq 1+x$ pour tout $x$, avec égalité si et
	seulement si $x=0$.
\end{enumerate}

\exercice{Fonction puissance \hfill \moyen}
\begin{enumerate}
	\item Écrire $3^{x}$ et $x^{x}$ ($x>0$) sous la forme $\e^{\ldots}$, puis
	calculer leur dérivée.
	\item Résoudre $3^{x}=5$ ; donner la valeur exacte puis une valeur
	approchée.
\end{enumerate}
