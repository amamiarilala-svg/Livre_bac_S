\chapter{Probabilités conditionnelles}

\begin{citationchapitre}
	« Lorsqu'une information nouvelle nous parvient, la probabilité que nous
	accordons à un événement doit être révisée. »\\[0.4em]
	— d'après Thomas Bayes
\end{citationchapitre}

\begin{objectifs}
	\begin{itemize}[label=\textbullet]
		\item Calculer une \textbf{probabilité conditionnelle} $\Prob_B(A)$.
		\item Construire et exploiter un \textbf{arbre pondéré}
		(probabilités composées).
		\item Appliquer la \textbf{formule des probabilités totales} sur un
		système complet.
		\item Appliquer la \textbf{formule de Bayes} (« inverser » un arbre).
		\item Reconnaître et utiliser l'\textbf{indépendance} de deux
		événements.
		\item Traiter des \textbf{tirages successifs} avec ou sans remise.
	\end{itemize}
\end{objectifs}

\section{Probabilité conditionnelle}
% ============================================================

\begin{definition}[Probabilité conditionnelle]
	Soient $A$ et $B$ deux événements avec $\Prob(B) \neq 0$. La
	\textbf{probabilité de $A$ sachant $B$}, notée $\Prob_B(A)$ ou
	$\Prob(A/B)$, est définie par
	\[
	\Prob_B(A) = \frac{\Prob(A\cap B)}{\Prob(B)}.
	\]
\end{definition}

\begin{center}
	\begin{tikzpicture}[>=stealth, font=\small]
		\draw (0,0) rectangle (5,3);
		\node[above right] at (0,3) {$\Omega$};
		\begin{scope}
			\clip (1.6,1.5) circle (1.1);
			\fill[black!30] (2.6,1.5) circle (1.1);
		\end{scope}
		\draw[thick] (2.6,1.5) circle (1.1);
		\draw (1.6,1.5) circle (1.1);
		\node at (1.2,1.5) {$A$};
		\node at (3.5,1.5) {$B$};
		\draw[->, thick] (5.8,1.5) -- (7.3,1.5);
		\node[above] at (6.5,1.9) {\small on se place dans $B$};
		\begin{scope}[xshift=9.3cm]
			\draw[thick] (0,0.4) circle (1.1);
			\node[above] at (0,1.6) {$B$};
			\fill[black!30] (0.4,0.4) circle (0.8);
			\draw (0.4,0.4) circle (0.8);
			\node at (-0.4,0.4) {$A\cap B$};
		\end{scope}
	\end{tikzpicture}
\end{center}

\begin{astucebac}
	Calculer $\Prob_B(A)$, c'est calculer la probabilité de $A$ en considérant
	que $B$ devient le nouvel univers de référence.
\end{astucebac}

\begin{propriete}{Formule des probabilités composées}{prop_composees}
	Pour tous événements $A$ et $B$ tels que $\Prob(B)\neq0$ :
	\[
	\Prob(A\cap B) = \Prob(B) \times \Prob_B(A).
	\]
	Plus généralement, si $\Prob(A_1\cap\cdots\cap A_{n-1})\neq 0$ :
	\[
	\Prob(A_1\cap\cdots\cap A_n)
	= \Prob(A_1)\,\Prob_{A_1}(A_2)\,\Prob_{A_1\cap A_2}(A_3)\cdots
	\]
\end{propriete}

\begin{exemple}
	Dans une classe, $60\,\%$ des élèves pratiquent un sport ($S$) et, parmi
	eux, $30\,\%$ pratiquent aussi la musique ($M$). Alors $\Prob(S) = 0{,}6$
	et $\Prob_S(M) = 0{,}3$, d'où
	$\Prob(S\cap M) = \Prob(S)\times \Prob_S(M) = 0{,}18$.
\end{exemple}

\section{Arbre pondéré}
% ============================================================

Un \textbf{arbre pondéré} représente une expérience à plusieurs étapes.
Chaque \textbf{chemin} correspond à une intersection d'événements.

\begin{center}
	\begin{tikzpicture}[>=stealth, grow=right, level distance=2.6cm, sibling distance=4.2cm, font=\small]
		\node (R) at (0,3) {}
		child { node (B1) {$B$} edge from parent node[above] {$\Prob(B)$} }
		child { node (B2) {$\overline{B}$} edge from parent node[below] {$\Prob(\overline{B})$} };
		\begin{scope}
			\path (B1) ++(2.6,1.5) node (A11) {$A$};
			\path (B1) ++(2.6,-1.5) node (A12) {$\overline{A}$};
			\draw[->] (B1) -- (A11) node[above, midway] {$\Prob_B(A)$};
			\draw[->] (B1) -- (A12) node[below, midway] {$\Prob_B(\overline{A})$};
			\path (B2) ++(2.6,1.5) node (A21) {$A$};
			\path (B2) ++(2.6,-1.5) node (A22) {$\overline{A}$};
			\draw[->] (B2) -- (A21) node[above, midway] {$\Prob_{\overline{B}}(A)$};
			\draw[->] (B2) -- (A22) node[below, midway] {$\Prob_{\overline{B}}(\overline{A})$};
		\end{scope}
	\end{tikzpicture}
\end{center}

\begin{methode}
	\textbf{Lire et utiliser un arbre pondéré :}
	\begin{enumerate}
		\item sur les branches issues d'un même nœud, la somme des
		probabilités vaut $1$ ;
		\item la probabilité d'un \textbf{chemin} est le \textbf{produit} des
		probabilités rencontrées le long de ce chemin ;
		\item la probabilité d'un événement est la \textbf{somme} des
		probabilités de tous les chemins qui y conduisent.
	\end{enumerate}
\end{methode}

\section{Probabilités totales, formule de Bayes}
% ============================================================

\begin{theoreme}[Formule des probabilités totales]
	Soit $\{A_1,A_2,\ldots,A_n\}$ un système complet d'événements de
	probabilités non nulles. Pour tout événement $B$ :
	\[
	\Prob(B) = \sum_{i=1}^{n}\Prob(A_i)\,\Prob_{A_i}(B).
	\]
	En particulier, avec le système complet $\{A,\overline{A}\}$ :
	\[
	\Prob(B) = \Prob(A)\,\Prob_A(B) + \Prob(\overline{A})\,\Prob_{\overline{A}}(B).
	\]
\end{theoreme}

\begin{center}
	\begin{tikzpicture}[>=stealth, font=\small]
		\draw (0,0) rectangle (6.5,3);
		\node[above right] at (0,3) {$\Omega$};
		\foreach \x in {1.3,2.6,3.9,5.2}{ \draw (\x,0) -- (\x,3); }
		\node at (0.65,2.6) {$A_1$};
		\node at (1.95,2.6) {$A_2$};
		\node at (3.25,2.6) {$A_3$};
		\node at (4.55,2.6) {$A_4$};
		\node at (5.85,2.6) {$A_5$};
		\draw[thick, fill=black!18, fill opacity=0.5]
		  (0.3,1.3) .. controls (2,0.4) and (4.5,2.2) .. (6.2,1.1)
		  -- (6.2,0.3) -- (0.3,0.3) -- cycle;
		\node at (3.2,0.75) {$B$};
	\end{tikzpicture}
\end{center}

\begin{theoreme}[Formule de Bayes]
	Soit $\{A_1,\ldots,A_n\}$ un système complet ($\Prob(A_i)\neq 0$) et $B$
	un événement avec $\Prob(B)\neq 0$. Pour tout $i$ :
	\[
	\Prob_B(A_i)
	= \frac{\Prob(A_i)\,\Prob_{A_i}(B)}
	       {\displaystyle\sum_{j=1}^{n}\Prob(A_j)\,\Prob_{A_j}(B)}.
	\]
\end{theoreme}

\begin{reperehisto}{Thomas Bayes (1702--1761)}
	Ce pasteur presbytérien anglais n'a jamais publié son théorème : son texte
	a été lu devant la Royal Society en 1763, deux ans après sa mort, par son
	ami Richard Price. La formule ne prendra sa portée générale qu'avec
	Laplace, qui la redécouvre indépendamment vers 1774.
\end{reperehisto}

\begin{methode}
	\textbf{« Inverser » un arbre (Bayes) :} on connaît les $\Prob_{A_i}(B)$
	(sens de l'arbre) et l'on cherche $\Prob_B(A_i)$ :
	\begin{enumerate}
		\item calculer $\Prob(A_i\cap B)=\Prob(A_i)\,\Prob_{A_i}(B)$ (chemin) ;
		\item calculer $\Prob(B)$ par la formule des probabilités totales ;
		\item faire le quotient
		$\Prob_B(A_i)=\dfrac{\Prob(A_i\cap B)}{\Prob(B)}$.
	\end{enumerate}
\end{methode}

\begin{exemple}
	Un atelier $A$ produit $70\,\%$ des pièces (dont $2\,\%$ défectueuses),
	l'atelier $B$ les $30\,\%$ restants (dont $5\,\%$ défectueuses).
	$\Prob(D)=0{,}7\times 0{,}02+0{,}3\times 0{,}05 = 0{,}029$, puis
	$\Prob_D(B)=\dfrac{0{,}3\times 0{,}05}{0{,}029}=\dfrac{0{,}015}{0{,}029}
	\approx 0{,}52$.
\end{exemple}

\begin{vraievie}{Un test positif, est-on vraiment malade ?}
	Sur le terrain, le paludisme se dépiste par un \textbf{TDR} (test de
	diagnostic rapide). Deux nombres caractérisent un tel test :
	\begin{itemize}[label=\textbullet]
		\item la \textbf{sensibilité} : probabilité qu'un malade soit détecté,
		ici $\Prob_M(T) = 0{,}95$ ;
		\item la \textbf{spécificité} : probabilité qu'un non-malade ait un
		test négatif, ici $\Prob_{\overline{M}}(\overline{T}) = 0{,}95$, donc
		$\Prob_{\overline{M}}(T) = 0{,}05$ (les « faux positifs »).
	\end{itemize}
	Dans un dispensaire où, en saison sèche, $2\,\%$ des consultants fiévreux
	sont réellement impaludés, $\Prob(M) = 0{,}02$. Probabilités totales :
	\[
	\Prob(T) = 0{,}02\times 0{,}95 + 0{,}98\times 0{,}05
	= 0{,}019 + 0{,}049 = 0{,}068,
	\]
	puis Bayes :
	\[
	\Prob_T(M) = \frac{0{,}019}{0{,}068} \approx 0{,}28.
	\]

	\begin{center}
		\begin{tikzpicture}[x=0.45cm, y=0.45cm, font=\scriptsize]
			\node at (9,1.6) {Les $68$ tests positifs, sur $1\,000$ consultants};
			\foreach \i in {0,...,18}{%
				\pgfmathtruncatemacro{\cx}{mod(\i,5)}%
				\pgfmathtruncatemacro{\cy}{-div(\i,5)}%
				\fill[black!75] (\cx,\cy) circle (0.3);
			}
			\foreach \i in {0,...,48}{%
				\pgfmathtruncatemacro{\cx}{mod(\i,13)+6}%
				\pgfmathtruncatemacro{\cy}{-div(\i,13)}%
				\draw[black!45] (\cx,\cy) circle (0.3);
			}
			\draw[decorate, decoration={brace, mirror, amplitude=4pt}]
			(-0.4,-3.7) -- (4.4,-3.7)
			node[midway, below=5pt] {$19$ malades};
			\draw[decorate, decoration={brace, mirror, amplitude=4pt}]
			(5.6,-3.7) -- (18.4,-3.7)
			node[midway, below=5pt] {$49$ non-malades};
		\end{tikzpicture}
	\end{center}

	Moins d'un test positif sur trois correspond donc à un malade --- et ce
	n'est pas la faute du test. Il ne se trompe que $5$ fois sur $100$, mais
	il s'applique à $980$ personnes saines : $5\,\%$ d'un grand groupe
	dépasse $95\,\%$ d'un petit.

	En zone de forte transmission, où $\Prob(M) = 0{,}4$, le même test donne
	$\Prob_T(M) \approx 0{,}93$ : la conclusion tient autant à la prévalence
	qu'à la qualité du test.

	\reserve{Sensibilité, spécificité et prévalence sont ici arrondies pour
	le calcul ; elles varient fortement selon le test employé, la région et
	la saison. C'est le raisonnement qu'il faut retenir, pas ces chiffres.}
\end{vraievie}

\section{Événements indépendants}
% ============================================================

\begin{definition}[Événements indépendants]
	Deux événements $A$ et $B$ sont \textbf{indépendants} lorsque
	\[
	\Prob(A\cap B) = \Prob(A)\times\Prob(B).
	\]
\end{definition}

\begin{propriete}{Caractérisation et stabilité}{prop_indep}
	Si $\Prob(B)\neq0$, alors $A$ et $B$ sont indépendants si et seulement si
	$\Prob_B(A) = \Prob(A)$ : la réalisation de $B$ ne modifie pas la
	probabilité de $A$.

	Si $A$ et $B$ sont indépendants, alors $A$ et $\overline{B}$,
	$\overline{A}$ et $B$, $\overline{A}$ et $\overline{B}$ le sont aussi.
\end{propriete}

\begin{attention}
	Ne pas confondre \textbf{indépendants}
	($\Prob(A\cap B)=\Prob(A)\Prob(B)$) et \textbf{incompatibles}
	($A\cap B=\emptyset$). Deux événements incompatibles de probabilités non
	nulles ne sont \textbf{jamais} indépendants.
\end{attention}

\begin{clindoeil}
	« Indépendants » et « incompatibles » ne sont pas synonymes : ce sont
	presque des contraires. Si deux événements ne peuvent pas se produire
	ensemble, apprendre que l'un s'est réalisé change tout pour l'autre --- on
	sait qu'il n'a pas eu lieu.
\end{clindoeil}

\section{Tirages successifs}
% ============================================================

Une urne contient $3$ boules rouges et $2$ boules bleues, indiscernables au
toucher. On tire successivement deux boules ; $R_i$ (resp. $B_i$) : « la
$i$-ème boule est rouge (resp. bleue) ».

\subsection{Tirages avec remise}

La boule est remise : la composition ne change pas, les deux tirages sont
\textbf{indépendants}.

\begin{center}
	\begin{tikzpicture}[>=stealth, grow=right, level distance=2.6cm, sibling distance=4.2cm, font=\small]
		\node (R) at (0,3) {}
		child { node (B1) {$R_1$} edge from parent node[above] {$\frac35$} }
		child { node (B2) {$B_1$} edge from parent node[below] {$\frac25$} };
		\begin{scope}
			\path (B1) ++(2.8,1.5) node (A11) {$R_2$};
			\path (B1) ++(2.8,-1.5) node (A12) {$B_2$};
			\draw[->] (B1) -- (A11) node[above, midway] {$\frac35$};
			\draw[->] (B1) -- (A12) node[below, midway] {$\frac25$};
			\path (B2) ++(2.8,1.5) node (A21) {$R_2$};
			\path (B2) ++(2.8,-1.5) node (A22) {$B_2$};
			\draw[->] (B2) -- (A21) node[above, midway] {$\frac35$};
			\draw[->] (B2) -- (A22) node[below, midway] {$\frac25$};
		\end{scope}
	\end{tikzpicture}
\end{center}

\subsection{Tirages sans remise}

La boule n'est pas remise : la composition change, les tirages sont
\textbf{dépendants}.

\begin{center}
	\begin{tikzpicture}[>=stealth, grow=right, level distance=2.6cm, sibling distance=4.2cm, font=\small]
		\node (R) at (0,3) {}
		child { node (B1) {$R_1$} edge from parent node[above] {$\frac35$} }
		child { node (B2) {$B_1$} edge from parent node[below] {$\frac25$} };
		\begin{scope}
			\path (B1) ++(2.8,1.5) node (A11) {$R_2$};
			\path (B1) ++(2.8,-1.5) node (A12) {$B_2$};
			\draw[->] (B1) -- (A11) node[above, midway] {$\frac24$};
			\draw[->] (B1) -- (A12) node[below, midway] {$\frac24$};
			\path (B2) ++(2.8,1.5) node (A21) {$R_2$};
			\path (B2) ++(2.8,-1.5) node (A22) {$B_2$};
			\draw[->] (B2) -- (A21) node[above, midway] {$\frac34$};
			\draw[->] (B2) -- (A22) node[below, midway] {$\frac14$};
		\end{scope}
	\end{tikzpicture}
\end{center}

\begin{exemple}
	Avec remise : $\Prob(R_1\cap R_2) = \dfrac35\times\dfrac35 = \dfrac{9}{25}$.

	Sans remise : $\Prob(R_1\cap R_2) = \dfrac35\times\dfrac24 = \dfrac{3}{10}$.
\end{exemple}

\section{Méthodes et exercices résolus}
% ============================================================

\begin{methode}
	\textbf{Traiter un énoncé « à deux caractères » :}
	\begin{enumerate}
		\item repérer le système complet naturel ($A/\overline{A}$, ou les
		machines, les urnes, les jours\ldots) ;
		\item construire l'arbre : premières branches = $\Prob(A_i)$, secondes
		branches = $\Prob_{A_i}(B)$ ;
		\item probabilités composées pour les chemins, probabilités totales
		pour $\Prob(B)$ ;
		\item Bayes si l'on demande une probabilité « dans l'autre sens ».
	\end{enumerate}
\end{methode}

\exerciceresolu
\textbf{Arbre, probabilités totales et Bayes}

Une usine produit des composants avec deux machines $M_1$ et $M_2$. $M_1$
fournit $65\,\%$ de la production avec un taux de défaut de $3\,\%$ ; $M_2$
fournit le reste avec un taux de défaut de $8\,\%$. On prélève un composant
au hasard ; $D$ : « il est défectueux ».
\begin{enumerate}
	\item Construire l'arbre pondéré.
	\item Calculer $\Prob(D)$.
	\item Le composant est défectueux : calculer la probabilité qu'il vienne
	de $M_2$.
\end{enumerate}

\solution

\textbf{1.} $\Prob(M_1)=0{,}65$, $\Prob(M_2)=0{,}35$,
$\Prob_{M_1}(D)=0{,}03$, $\Prob_{M_2}(D)=0{,}08$.

\begin{center}
	\begin{tikzpicture}[>=stealth, grow=right, level distance=2.6cm, sibling distance=3.8cm, font=\small]
		\node (R) at (0,3) {}
		child { node (B1) {$M_1$} edge from parent node[above] {$0{,}65$} }
		child { node (B2) {$M_2$} edge from parent node[below] {$0{,}35$} };
		\begin{scope}
			\path (B1) ++(2.6,1.3) node (A11) {$D$};
			\path (B1) ++(2.6,-1.3) node (A12) {$\overline{D}$};
			\draw[->] (B1) -- (A11) node[above, midway] {$0{,}03$};
			\draw[->] (B1) -- (A12) node[below, midway] {$0{,}97$};
			\path (B2) ++(2.6,1.3) node (A21) {$D$};
			\path (B2) ++(2.6,-1.3) node (A22) {$\overline{D}$};
			\draw[->] (B2) -- (A21) node[above, midway] {$0{,}08$};
			\draw[->] (B2) -- (A22) node[below, midway] {$0{,}92$};
		\end{scope}
	\end{tikzpicture}
\end{center}

\textbf{2.} $\Prob(D) = 0{,}65\times 0{,}03 + 0{,}35\times 0{,}08
= 0{,}0195 + 0{,}028 = 0{,}0475$.

\textbf{3.} $\Prob_D(M_2) = \dfrac{\Prob(M_2\cap D)}{\Prob(D)}
= \dfrac{0{,}028}{0{,}0475} \approx 0{,}589$.

\exerciceresolu
\textbf{Tirages sans remise}

Une urne contient $4$ boules vertes et $3$ jaunes, indiscernables au toucher.
On tire successivement et sans remise deux boules.
\begin{enumerate}
	\item Calculer la probabilité de tirer deux boules de la même couleur.
	\item Calculer la probabilité de tirer au moins une verte.
\end{enumerate}

\solution

\textbf{1.} $\Prob(V_1\cap V_2)=\dfrac47\times\dfrac36=\dfrac{12}{42}$ et
$\Prob(J_1\cap J_2)=\dfrac37\times\dfrac26=\dfrac{6}{42}$. Ces deux événements
étant incompatibles, la probabilité cherchée est
$\dfrac{12}{42}+\dfrac{6}{42}=\dfrac{18}{42}=\dfrac37$.

\textbf{2.} Contraire : « aucune verte » $=J_1\cap J_2$, de probabilité
$\dfrac{6}{42}=\dfrac17$. Donc $\Prob(\text{au moins une verte})
=1-\dfrac17=\dfrac67$.

\exerciceresolu
\textbf{Indépendance}

On lance un dé équilibré. $A$ : « obtenir un nombre pair », $B$ : « obtenir
un multiple de $3$ », $C$ : « obtenir un nombre $\leqslant 4$ ».
\begin{enumerate}
	\item $A$ et $B$ sont-ils indépendants ?
	\item $A$ et $C$ sont-ils indépendants ?
\end{enumerate}

\solution

$\Prob(A)=\dfrac12$, $\Prob(B)=\dfrac13$, $\Prob(C)=\dfrac23$.

\textbf{1.} $A\cap B=\{6\}$, $\Prob(A\cap B)=\dfrac16=\dfrac12\times\dfrac13
=\Prob(A)\Prob(B)$ : \textbf{indépendants}.

\textbf{2.} $A\cap C=\{2,4\}$, $\Prob(A\cap C)=\dfrac13$, alors que
$\Prob(A)\Prob(C)=\dfrac12\times\dfrac23=\dfrac13$ : \textbf{indépendants}
également.

\exercice{Lire un arbre pondéré \hfill \facile}

			\path (B2) ++(2.6,1.1) node (A21) {$B$};

			\draw[->] (B2) -- (A21) node[above, midway] {$0{,}5$};

		\end{scope}
	\end{tikzpicture}

\begin{enumerate}
	\item Donner $\Prob(A)$ et $\Prob_A(B)$.
	\item Calculer $\Prob(A\cap B)$ et $\Prob(\overline{A}\cap B)$.
	\item En déduire $\Prob(B)$.
	\item Calculer $\Prob_B(A)$.
\end{enumerate}

\exercice{Vrai ou faux \hfill \facile}
Répondre par \emph{vrai} ou \emph{faux} en justifiant.
\begin{enumerate}
	\item On a toujours $\Prob_B(A)=\Prob_A(B)$.
	\item Si $A$ et $B$ sont indépendants, alors $\Prob_B(A)=\Prob(A)$.
	\item Si $\Prob_B(A)=\Prob(A)$, alors $A$ et $B$ sont incompatibles.
	\item Sur un arbre pondéré, la somme des probabilités des branches issues
	d'un même nœud vaut $1$.
	\item $\Prob(A\cap B)\leqslant\Prob(B)$.
	\item Si $\Prob(B)=0{,}2$ et $\Prob(A\cap B)=0{,}1$, alors
	$\Prob_B(A)=0{,}5$.
	\item Deux événements incompatibles de probabilités non nulles sont
	dépendants.
\end{enumerate}

\exercice{Conditionnelles directes \hfill \facile}
Dans un lycée, $55\,\%$ des élèves sont des filles ($F$). Parmi les filles,
$40\,\%$ étudient les sciences ($S$) ; parmi les garçons, $60\,\%$.
\begin{multicols}{2}
\begin{enumerate}
	\item $\Prob(F)$, $\Prob_F(S)$, $\Prob_{\overline{F}}(S)$.
	\item $\Prob(F\cap S)$.
	\item $\Prob(\overline{F}\cap S)$.
	\item $\Prob(S)$.
\end{enumerate}
\end{multicols}

\exercice{Météo et retard \hfill \moyen}
Il pleut ($R$) un jour sur quatre. S'il pleut, le bus est en retard avec
probabilité $0{,}5$ ; sinon avec probabilité $0{,}1$.
\begin{enumerate}
	\item Construire l'arbre pondéré.
	\item Calculer la probabilité que le bus soit en retard un jour donné.
	\item Le bus est en retard : calculer la probabilité qu'il pleuve.
\end{enumerate}

\exercice{Trois urnes \hfill \moyen}
On choisit au hasard l'une des trois urnes $U_1$ ($2$ blanches, $3$ noires),
$U_2$ ($4$ blanches, $1$ noire), $U_3$ ($3$ blanches, $3$ noires), puis on y
tire une boule.
\begin{enumerate}
	\item Calculer la probabilité de tirer une boule blanche.
	\item La boule tirée est blanche : quelle est la probabilité qu'elle
	vienne de $U_2$ ?
\end{enumerate}

\exercice{Indépendance \hfill \moyen}
On lance deux dés équilibrés. $A$ : « la somme vaut $7$ », $B$ : « le premier
dé donne $3$ », $C$ : « les deux dés donnent le même résultat ».
\begin{multicols}{2}
\begin{enumerate}
	\item $\Prob(A)$, $\Prob(B)$, $\Prob(C)$.
	\item $A$ et $B$ indépendants ?
	\item $A$ et $C$ indépendants ?
	\item $B$ et $C$ indépendants ?
\end{enumerate}
\end{multicols}

\exercice{Sujet type Bac — test de dépistage \hfill \difficile}
Une maladie touche $2\,\%$ d'une population. Un test de dépistage donne un
résultat positif dans $95\,\%$ des cas si la personne est malade
(\emph{sensibilité}), et négatif dans $90\,\%$ des cas si elle est saine
(\emph{spécificité}). On note $M$ : « être malade » et $T$ : « le test est
positif ».

\textbf{Partie A}
\begin{enumerate}
	\item Traduire les données et construire l'arbre pondéré.
	\item Calculer $\Prob(M\cap T)$ et $\Prob(\overline{M}\cap T)$.
	\item En déduire $\Prob(T)$.
\end{enumerate}

\textbf{Partie B}
\begin{enumerate}
	\setcounter{enumi}{3}
	\item Une personne a un test positif. Calculer la probabilité qu'elle soit
	réellement malade (\emph{valeur prédictive positive}). Commenter.
	\item Une personne a un test négatif. Calculer la probabilité qu'elle soit
	tout de même malade.
	\item On refait passer le test, indépendamment, aux personnes ayant eu un
	premier test positif. Calculer la probabilité qu'une personne saine ait
	deux tests positifs.
\end{enumerate}

\exercice{Sujet type Bac — chaîne de production \hfill \difficile}
Trois machines $M_1$, $M_2$, $M_3$ produisent respectivement $50\,\%$,
$30\,\%$ et $20\,\%$ des pièces d'une usine, avec des taux de défaut de
$4\,\%$, $5\,\%$ et $10\,\%$. On prélève une pièce au hasard ; $D$ : « elle
est défectueuse ».
\begin{enumerate}
	\item Représenter la situation par un arbre pondéré.
	\item Montrer que $\Prob(D)=0{,}055$.
	\item La pièce est défectueuse. Déterminer la machine la plus probablement
	à l'origine du défaut, en calculant $\Prob_D(M_1)$, $\Prob_D(M_2)$,
	$\Prob_D(M_3)$.
	\item On prélève $2$ pièces au hasard, de façon indépendante. Calculer la
	probabilité qu'aucune ne soit défectueuse, puis qu'au moins une le soit.
\end{enumerate}

\exercice{Sujet type Bac — marche aléatoire \hfill \difficile}
Un jeton se déplace entre deux cases $E$ et $F$. À chaque étape : s'il est en
$E$, il passe en $F$ avec probabilité $\dfrac16$ ; s'il est en $F$, il passe
en $E$ avec probabilité $\dfrac14$. Au départ ($n=0$), le jeton est en $E$.
On note $p_n$ la probabilité qu'il soit en $E$ à l'étape $n$.
\begin{enumerate}
	\item Donner $p_0$. À l'aide d'un arbre, exprimer $p_{n+1}$ en fonction de
	$p_n$. Montrer que $p_{n+1}=\dfrac{7}{12}\,p_n+\dfrac14$.
	\item On pose $q_n=p_n-\dfrac{3}{5}$. Montrer que $(q_n)$ est géométrique
	de raison $\dfrac{7}{12}$.
	\item En déduire l'expression de $p_n$ en fonction de $n$.
	\item Déterminer $\displaystyle\lim_{n\to+\infty}p_n$ et interpréter.
\end{enumerate}

\exercice{Sujet type Bac — deux tirages, avec puis sans remise \hfill \difficile}
Une urne contient $5$ boules rouges et $3$ vertes, indiscernables au toucher.

\textbf{Partie A — avec remise}
On tire une boule, on la remet, on en tire une seconde.
\begin{enumerate}
	\item Calculer la probabilité de tirer deux rouges.
	\item Calculer la probabilité de tirer exactement une rouge.
	\item Les événements « la $1^{\text{re}}$ est rouge » et « la
	$2^{\text{e}}$ est rouge » sont-ils indépendants ?
\end{enumerate}

\textbf{Partie B — sans remise}
On tire successivement deux boules sans remise.
\begin{enumerate}
	\setcounter{enumi}{3}
	\item Construire l'arbre pondéré.
	\item Calculer la probabilité que la $2^{\text{e}}$ boule soit verte.
	\item La $2^{\text{e}}$ boule est verte : calculer la probabilité que la
	$1^{\text{re}}$ ait été rouge.
\end{enumerate}

\exercice{Sujet type Bac — QCM et indépendance \hfill \difficile}
Un élève répond à deux questions indépendantes. Il connaît la réponse de la
première avec probabilité $0{,}7$, celle de la seconde avec probabilité
$0{,}6$. S'il connaît la réponse, il répond juste ; sinon il répond au hasard
et a $1$ chance sur $4$ d'être juste.
\begin{enumerate}
	\item Calculer la probabilité qu'il réponde juste à la première question.
	\item Calculer de même pour la seconde question.
	\item En déduire la probabilité qu'il réponde juste aux deux questions.
	\item Calculer la probabilité qu'il réponde juste à exactement une des
	deux.
	\item Il a répondu juste à la première question. Calculer la probabilité
	qu'il en connaissait réellement la réponse.
\end{enumerate}
