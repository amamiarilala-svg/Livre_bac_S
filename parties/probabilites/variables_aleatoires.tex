\chapter{Variables aléatoires}

\begin{citationchapitre}
	« L'espérance mathématique est la somme de chaque gain multiplié par la
	probabilité de l'obtenir. »\\[0.4em]
	— Christiaan Huygens, \emph{De ratiociniis in ludo aleae}
\end{citationchapitre}

\begin{objectifs}
	\begin{itemize}[label=\textbullet]
		\item Déterminer la \textbf{loi de probabilité} d'une variable
		aléatoire.
		\item Déterminer et représenter une \textbf{fonction de répartition}.
		\item Calculer et interpréter l'\textbf{espérance}, la
		\textbf{variance} et l'\textbf{écart-type} (formule de
		König-Huygens).
		\item Utiliser la linéarité : $\Esp(aX+b)$, $\Var(aX+b)$.
		\item Décider si un \textbf{jeu} est équitable, favorable ou
		défavorable.
		\item Reconnaître une \textbf{épreuve} et un \textbf{schéma de
		Bernoulli}.
	\end{itemize}
\end{objectifs}

On considère, comme fil rouge, l'expérience suivante : on lance deux pièces
équilibrées et on note $X$ le nombre de « Pile » obtenus.

\section{Variable aléatoire et loi de probabilité}
% ============================================================

\begin{definition}[Variable aléatoire]
	Définir une \textbf{variable aléatoire} $X$ sur un univers $\Omega$, c'est
	associer un nombre réel à chaque issue de $\Omega$. L'ensemble des valeurs
	prises par $X$ est noté $X(\Omega)$.
\end{definition}

\begin{exemple}
	Ici $\Omega = \{PP,\,PF,\,FP,\,FF\}$ et $X(\Omega) = \{0,1,2\}$.
\end{exemple}

\begin{definition}[Loi de probabilité]
	La \textbf{loi de probabilité} de $X$ associe à chaque valeur $x_i$ de
	$X(\Omega)$ la probabilité $p_i = \Prob(X=x_i)$. On a toujours
	$\sum_i p_i = 1$.
\end{definition}

\begin{exemple}
	Les quatre issues étant équiprobables :
	\begin{center}
		\begin{tabular}{|c|c|c|c|}
			\hline
			$x_i$ & $0$ & $1$ & $2$ \\
			\hline
			$\Prob(X=x_i)$ & $\dfrac14$ & $\dfrac12$ & $\dfrac14$ \\
			\hline
		\end{tabular}
	\end{center}
\end{exemple}

\begin{center}
	\begin{tikzpicture}[>=stealth, font=\small]
		\draw[->] (-0.6,0) -- (3.5,0) node[right] {$x_i$};
		\draw[->] (0,-0.3) -- (0,2.6) node[above] {$\Prob(X=x_i)$};
		\foreach \x/\h in {0/1, 1/2, 2/1}{
			\draw[very thick] (\x,0) -- (\x,\h);
			\fill (\x,\h) circle (1.6pt);
			\node[below] at (\x,0) {$\x$};
		}
		\draw[dashed, gray] (0,1) -- (2,1) (0,2) -- (1,2);
		\node[left] at (0,1) {\scriptsize $\frac14$};
		\node[left] at (0,2) {\scriptsize $\frac12$};
	\end{tikzpicture}
\end{center}

\section{Fonction de répartition}
% ============================================================

\begin{definition}[Fonction de répartition]
	La \textbf{fonction de répartition} de $X$ est la fonction $F$ définie sur
	$\R$ par $F(x) = \Prob(X \leqslant x)$. C'est une fonction
	\textbf{croissante}, en escalier, nulle pour $x$ assez petit et égale à
	$1$ pour $x$ assez grand.
\end{definition}

\begin{exemple}
	Pour le fil rouge : $F(x) = 0$ si $x<0$ ; $F(x) = \dfrac14$ si
	$0\leqslant x<1$ ; $F(x)=\dfrac34$ si $1\leqslant x<2$ ; $F(x)=1$ si
	$x\geqslant2$.
\end{exemple}

\begin{center}
	\begin{tikzpicture}[>=stealth, font=\small]
		\draw[->] (-1,0) -- (4,0) node[right] {$x$};
		\draw[->] (0,-0.3) -- (0,3.4) node[above] {$F(x)$};
		\draw[thick] (-1,0) -- (0,0);
		\draw[thick] (0,0.75) -- (1,0.75);
		\draw[thick] (1,2.25) -- (2,2.25);
		\draw[thick] (2,3) -- (3.5,3);
		\draw[dashed, gray] (0,0.75) -- (1,0.75) (0,2.25) -- (1,2.25) (0,3) -- (2,3);
		\draw[fill=black] (0,0.75) circle (1.6pt);
		\draw[fill=white] (1,0.75) circle (1.6pt);
		\draw[fill=black] (1,2.25) circle (1.6pt);
		\draw[fill=white] (2,2.25) circle (1.6pt);
		\draw[fill=black] (2,3) circle (1.6pt);
		\node[below] at (0,0) {$0$};
		\node[below] at (1,0) {$1$};
		\node[below] at (2,0) {$2$};
		\node[left] at (0,0.75) {\scriptsize $\frac14$};
		\node[left] at (0,2.25) {\scriptsize $\frac34$};
		\node[left] at (0,3) {\scriptsize $1$};
	\end{tikzpicture}
\end{center}

\begin{astucebac}
	$F$ « saute » de $p_i$ exactement en $x=x_i$, et le point plein (valeur
	incluse) est toujours \emph{à droite} du saut. Réciproquement,
	$\Prob(X=x_i)$ est la hauteur du saut de $F$ en $x_i$, et
	$\Prob(a<X\leqslant b)=F(b)-F(a)$.
\end{astucebac}

\section{Espérance, variance, écart-type}
% ============================================================

\begin{definition}[Espérance mathématique]
	L'\textbf{espérance} de $X$ est le nombre
	\[
	\Esp(X) = \sum_i p_i\,x_i.
	\]
	Elle représente la valeur \textbf{moyenne} de $X$ obtenue en répétant un
	grand nombre de fois l'expérience.
\end{definition}

\begin{definition}[Variance et écart-type]
	La \textbf{variance} de $X$ est
	\[
	\Var(X) = \sum_i p_i\bigl(x_i-\Esp(X)\bigr)^2
	= \left(\sum_i p_i x_i^2\right) - \bigl[\Esp(X)\bigr]^2
	\qquad\text{(formule de König-Huygens).}
	\]
	L'\textbf{écart-type} est $\sigma(X) = \sqrt{\Var(X)}$. Tous deux mesurent
	la \textbf{dispersion} de $X$ autour de $\Esp(X)$.
\end{definition}

\begin{methode}
	\textbf{Calculer $\Esp(X)$, $\Var(X)$, $\sigma(X)$ :}
	\begin{enumerate}
		\item dresser le tableau de la loi de $X$ ;
		\item $\Esp(X) = \sum p_i x_i$ ;
		\item $\sum p_i x_i^2$, puis $\Var(X) = \sum p_i x_i^2 - [\Esp(X)]^2$ ;
		\item $\sigma(X) = \sqrt{\Var(X)}$.
	\end{enumerate}
\end{methode}

\begin{exemple}
	Pour le fil rouge : $\Esp(X) = \dfrac14\cdot0 + \dfrac12\cdot1 +
	\dfrac14\cdot2 = 1$ ;
	$\sum p_i x_i^2 = \dfrac12\cdot1 + \dfrac14\cdot4 = \dfrac32$ ;
	$\Var(X) = \dfrac32 - 1 = \dfrac12$ ; $\sigma(X) = \dfrac{\sqrt2}{2}$.
\end{exemple}

\begin{propriete}{Linéarité}{prop_lin}
	Pour tous réels $a$ et $b$ :
	\[
	\Esp(aX+b) = a\,\Esp(X)+b, \qquad
	\Var(aX+b) = a^2\,\Var(X), \qquad
	\sigma(aX+b) = |a|\,\sigma(X).
	\]
	De plus, pour toute fonction $g$ :
	$\Esp\bigl(g(X)\bigr)=\sum_i p_i\,g(x_i)$.
\end{propriete}

\begin{definition}[Jeu équitable]
	Dans un jeu où $X$ désigne le gain algébrique d'un joueur, le jeu est dit
	\textbf{équitable} si $\Esp(X)=0$, \textbf{favorable} au joueur si
	$\Esp(X)>0$, \textbf{défavorable} si $\Esp(X)<0$.
\end{definition}

\begin{astucebac}
	Pour rendre un jeu équitable, on cherche souvent la \textbf{mise} $m$ ou
	le \textbf{gain} $g$ tel que $\Esp(X)=0$, où $X$ contient un terme
	$-m$ (mise payée) ou $+g$.
\end{astucebac}

\section{Épreuve et schéma de Bernoulli}
% ============================================================

\begin{definition}[Épreuve de Bernoulli]
	Une \textbf{épreuve de Bernoulli} est une expérience à deux issues : un
	\textbf{succès} (probabilité $p$) et un \textbf{échec} (probabilité
	$1-p$). La variable $X$ valant $1$ en cas de succès et $0$ sinon suit la
	\textbf{loi de Bernoulli} de paramètre $p$, notée $\mathscr{B}(p)$.
\end{definition}


\begin{center}
	\begin{tikzpicture}[>=stealth, grow=right, level distance=3cm, sibling distance=1.6cm, font=\small]
		\node {}
		child { node {$0$ (échec)} edge from parent node[below] {$1-p$} }
		child { node {$1$ (succès)} edge from parent node[above] {$p$} };
	\end{tikzpicture}
\end{center}

\begin{propriete}{Espérance et variance d'une loi de Bernoulli}{prop_bern}
	Si $X\sim\mathscr{B}(p)$, alors $\Esp(X) = p$ et $\Var(X) = p(1-p)$.
\end{propriete}

\begin{definition}[Schéma de Bernoulli]
	Un \textbf{schéma de Bernoulli} de paramètres $n$ et $p$ consiste à
	répéter $n$ fois, de façon \textbf{identique et indépendante}, la même
	épreuve de Bernoulli de paramètre $p$.
\end{definition}

	binomiale} (chapitre suivant).
\end{exemple}

\section{Méthodes et exercices résolus}
% ============================================================

\begin{methode}
	\textbf{Étudier un jeu :}
	\begin{enumerate}
		\item identifier la variable $X$ (gain \emph{algébrique} : compter la
		mise en négatif) ;
		\item déterminer $X(\Omega)$ et la loi (arbre, dénombrement,
		équiprobabilité) ;
		\item calculer $\Esp(X)$ et conclure (équitable / favorable) ;
		\item si demandé, calculer $\Var(X)$, $\sigma(X)$, ou résoudre
		$\Esp(X)=0$ en la mise.
	\end{enumerate}
\end{methode}

\exerciceresolu
\textbf{Loi d'un gain, espérance, variance}

Un jeu consiste à lancer un dé équilibré à six faces. Obtenir $6$ rapporte
$10$ points ; obtenir $3$ rapporte $2$ points ; tout autre résultat fait
perdre $1$ point. On note $X$ le gain algébrique.
\begin{enumerate}
	\item Déterminer la loi de $X$.
	\item Calculer $\Esp(X)$. Le jeu est-il favorable au joueur ?
	\item Calculer $\Var(X)$ et $\sigma(X)$.
\end{enumerate}

\solution

\textbf{1.} $X(\Omega) = \{-1,2,10\}$ avec $\Prob(X=10)=\dfrac16$,
$\Prob(X=2)=\dfrac16$, $\Prob(X=-1)=\dfrac46=\dfrac23$.
\begin{center}
	\begin{tabular}{|c|c|c|c|}
		\hline
		$x_i$ & $-1$ & $2$ & $10$ \\
		\hline
		$\Prob(X=x_i)$ & $\dfrac23$ & $\dfrac16$ & $\dfrac16$ \\
		\hline
	\end{tabular}
\end{center}

\textbf{2.} $\Esp(X) = \dfrac23(-1) + \dfrac16\cdot2 + \dfrac16\cdot10
= -\dfrac23+\dfrac13+\dfrac53 = \dfrac43 > 0$ : jeu favorable au joueur.

\textbf{3.} $\displaystyle\sum p_i x_i^2 = \dfrac23\cdot1 + \dfrac16\cdot4 +
\dfrac16\cdot100 = \dfrac{4+4+100}{6} = 18$, donc
\[
\Var(X) = 18 - \left(\frac43\right)^2 = 18-\frac{16}{9} = \frac{146}{9},
\qquad \sigma(X) = \frac{\sqrt{146}}{3}\approx 4{,}03.
\]

\exerciceresolu
\textbf{Déterminer une mise équitable}

À une loterie, un billet coûte $m$ ariary. Sur $100$ billets, $1$ rapporte
$500$~Ar, $9$ rapportent $100$~Ar, les autres ne rapportent rien. On note $X$
le gain algébrique de l'acheteur d'un billet.
\begin{enumerate}
	\item Exprimer la loi de $X$ en fonction de $m$.
	\item Déterminer $m$ pour que le jeu soit équitable.
\end{enumerate}

\solution

\textbf{1.} $X$ prend les valeurs $500-m$ (proba $\frac{1}{100}$),
$100-m$ (proba $\frac{9}{100}$) et $-m$ (proba $\frac{90}{100}$).

\textbf{2.}
\[
\Esp(X) = \frac{500-m}{100} + \frac{9(100-m)}{100} + \frac{90(-m)}{100}
= \frac{500 + 900 - 100m}{100} = \frac{1400}{100} - m = 14 - m.
\]
Le jeu est équitable pour $\Esp(X)=0$, soit $m = 14$~Ar.

\exerciceresolu
\textbf{Fonction de répartition}

$X$ prend les valeurs $-2$, $0$, $3$ avec les probabilités $0{,}2$ ; $0{,}5$ ;
$0{,}3$.
\begin{enumerate}
	\item Déterminer $F$, fonction de répartition de $X$.
	\item Calculer $\Prob(-2 < X \leqslant 3)$ à l'aide de $F$.
	\item Calculer $\Esp(X)$ et $\sigma(X)$.
\end{enumerate}

\solution

\textbf{1.} $F(x)=0$ si $x<-2$ ; $0{,}2$ si $-2\leqslant x<0$ ; $0{,}7$ si
$0\leqslant x<3$ ; $1$ si $x\geqslant3$.

\textbf{2.} $\Prob(-2<X\leqslant3) = F(3)-F(-2) = 1 - 0{,}2 = 0{,}8$.

\textbf{3.} $\Esp(X) = 0{,}2(-2)+0{,}5\cdot0+0{,}3\cdot3 = -0{,}4+0{,}9
= 0{,}5$.
$\sum p_i x_i^2 = 0{,}2\cdot4 + 0{,}3\cdot9 = 0{,}8+2{,}7 = 3{,}5$, donc
$\Var(X)=3{,}5-0{,}25 = 3{,}25$ et $\sigma(X)=\sqrt{3{,}25}\approx1{,}80$.

\exercice{Vrai ou faux \hfill \facile}
Répondre par \emph{vrai} ou \emph{faux} en justifiant.
\begin{enumerate}
	\item L'espérance d'une variable aléatoire est toujours une valeur prise
	par cette variable.
	\item Si $\Esp(X)=0$, alors $X$ est la variable nulle.
	\item $\Var(X)\geqslant 0$ pour toute variable aléatoire $X$.
	\item $\Esp(2X+3)=2\Esp(X)+3$.
	\item $\Var(2X+3)=2\Var(X)+3$.
	\item Une fonction de répartition est croissante et à valeurs dans
	$[0\,;1]$.
	\item Si $X\sim\mathscr{B}(p)$, alors $\Var(X)=p(1-p)$ est maximale pour
	$p=\dfrac12$.
\end{enumerate}

\exercice{Loi et espérance \hfill \facile}
On lance un dé équilibré ; $X$ vaut le numéro obtenu s'il est pair, et $0$
sinon.
\begin{multicols}{2}
\begin{enumerate}
	\item Déterminer $X(\Omega)$ et la loi de $X$.
	\item Représenter la loi par un diagramme en bâtons.
	\item Calculer $\Esp(X)$.
	\item Calculer $\Var(X)$ et $\sigma(X)$.
\end{enumerate}
\end{multicols}

\exercice{Urne et gain \hfill \moyen}
Une urne contient $2$ boules rouges, $3$ vertes et $5$ jaunes. Une rouge
rapporte $5$ points, une verte $1$ point, une jaune fait perdre $2$ points.
$X$ = gain.
\begin{enumerate}
	\item Déterminer la loi de $X$.
	\item Calculer $\Esp(X)$. Le jeu est-il équitable ?
	\item Calculer $\Var(X)$ et $\sigma(X)$.
\end{enumerate}

\exercice{Transformation affine \hfill \moyen}
Une variable $X$ vérifie $\Esp(X)=4$ et $\Var(X)=9$. On pose $Y=3X-5$ et
$Z=\dfrac{X-4}{3}$.
\begin{multicols}{2}
\begin{enumerate}
	\item Calculer $\Esp(Y)$ et $\sigma(Y)$.
	\item Calculer $\Esp(Z)$ et $\Var(Z)$.
\end{enumerate}
\end{multicols}

\exercice{Tirage sans remise et loi \hfill \difficile}

	\item Déterminer la fonction de répartition de $X$.
	\item Calculer $\Esp(X)$ et $\Var(X)$.
\end{enumerate}

\exercice{Sujet type Bac — roue de loterie \hfill \difficile}
Une roue est partagée en $8$ secteurs identiques : $1$ secteur rapporte
$20$ points, $2$ secteurs rapportent $5$ points, $5$ secteurs ne rapportent
rien. Pour jouer, on mise $3$ points. On note $X$ le gain algébrique du
joueur (gain obtenu moins la mise).

\textbf{Partie A}
\begin{enumerate}
	\item Déterminer $X(\Omega)$.
	\item Établir la loi de probabilité de $X$.
	\item Calculer $\Esp(X)$. Le jeu est-il favorable au joueur ?
\end{enumerate}

\textbf{Partie B}
\begin{enumerate}
	\setcounter{enumi}{3}
	\item Calculer $\Var(X)$ et $\sigma(X)$ (arrondir à $10^{-2}$).
	\item L'organisateur souhaite un jeu équitable en modifiant seulement la
	mise $m$. Exprimer $\Esp(X)$ en fonction de $m$ et déterminer $m$.
	\item Avec la mise initiale de $3$ points, un joueur joue $50$ parties
	indépendantes. Estimer son gain algébrique total moyen.
\end{enumerate}

\exercice{Sujet type Bac — contrat d'assurance \hfill \difficile}
Une compagnie assure un objet contre le vol. La probabilité de vol dans
l'année est $0{,}03$. En cas de vol, la compagnie verse $2\,000\,000$~Ar. La
prime annuelle payée par le client est $P$. On note $Y$ le résultat
algébrique de la compagnie pour un contrat (recette moins dépense).
\begin{enumerate}
	\item Déterminer la loi de $Y$ en fonction de $P$.
	\item Exprimer $\Esp(Y)$ en fonction de $P$.
	\item Déterminer la prime $P_0$ rendant le contrat équitable.
	\item La compagnie fixe $P = 90\,000$~Ar. Calculer son bénéfice moyen par
	contrat.
	\item La compagnie a $10\,000$ contrats identiques et indépendants.
	Estimer son bénéfice annuel moyen total.
	\item Calculer $\sigma(Y)$ pour un contrat (arrondir à $10^{3}$~Ar près).
\end{enumerate}

\exercice{Sujet type Bac — deux dés et une variable \hfill \difficile}
On lance deux dés équilibrés et on note $X$ la valeur absolue de la
différence des deux résultats.

\textbf{Partie A}
\begin{enumerate}
	\item Déterminer $X(\Omega)$.
	\item Montrer que $\Prob(X=0)=\dfrac16$ et calculer $\Prob(X=1)$.
	\item Établir la loi de probabilité complète de $X$.
\end{enumerate}

\textbf{Partie B}
\begin{enumerate}
	\setcounter{enumi}{3}
	\item Calculer $\Esp(X)$.
	\item Déterminer et représenter la fonction de répartition de $X$.
	\item On gagne $100$~Ar si $X\leqslant 1$, on perd $50$~Ar sinon. Ce jeu
	est-il favorable au joueur ?
\end{enumerate}

\exercice{Sujet type Bac — comparer deux jeux \hfill \difficile}
Deux jeux proposent les gains algébriques suivants.
\begin{center}
	\begin{tabular}{|c|c|c|c|}
		\hline
		$x_i$ (jeu $A$) & $-10$ & $0$ & $20$ \\ \hline
		$\Prob$ & $0{,}5$ & $0{,}3$ & $0{,}2$ \\ \hline
	\end{tabular}
	\qquad
	\begin{tabular}{|c|c|c|c|}
		\hline
		$y_i$ (jeu $B$) & $-5$ & $0$ & $5$ \\ \hline
		$\Prob$ & $0{,}3$ & $0{,}6$ & $0{,}1$ \\ \hline
	\end{tabular}
\end{center}
\begin{enumerate}
	\item Calculer $\Esp(A)$ et $\Esp(B)$. Que constate-t-on ?
	\item Calculer $\Var(A)$, $\Var(B)$, puis $\sigma(A)$ et $\sigma(B)$.
	\item Un joueur veut limiter le risque de grosses pertes : quel jeu lui
	conseiller ? Justifier avec $\sigma$.
	\item Pour rendre le jeu $A$ équitable, on ajoute une prime $c$ à chaque
	gain. Déterminer $c$, puis donner l'espérance et la variance du jeu ainsi
	modifié.
\end{enumerate}

\exercice{Sujet type Bac — épreuve de Bernoulli et répétition \hfill \difficile}
Une urne contient $8$ boules : $5$ blanches et $3$ noires, indiscernables au
toucher. On tire une boule, succès = « blanche », et on note $Y$ la variable
de Bernoulli associée.
\begin{enumerate}
	\item Justifier qu'il s'agit d'une épreuve de Bernoulli et donner son
	paramètre $p$.
	\item Donner $\Esp(Y)$ et $\Var(Y)$.
	\item On répète l'épreuve $4$ fois \emph{avec remise}. Justifier qu'il
	s'agit d'un schéma de Bernoulli et préciser ses paramètres.
	\item À l'aide d'un arbre, calculer la probabilité d'obtenir exactement
	$2$ boules blanches sur les $4$ tirages.
	\item On répète maintenant l'épreuve $4$ fois \emph{sans remise}.
	Est-ce encore un schéma de Bernoulli ? Justifier.
\end{enumerate}
