% ============================================================
\section{Définition et existence du barycentre}
% ============================================================

\begin{definition}
	Soient $A_1, A_2, \ldots, A_n$ des points du plan et $\alpha_1, \alpha_2, \ldots, \alpha_n$
	des réels tels que $\alpha_1 + \alpha_2 + \cdots + \alpha_n \neq 0$.

	Il existe un unique point $G$ vérifiant :
	$$
	\alpha_1\,\vect{GA_1} + \alpha_2\,\vect{GA_2} + \cdots + \alpha_n\,\vect{GA_n} = \vect{0}.
	$$
	Ce point $G$ s'appelle le \textbf{barycentre} du système de points pondérés
	$$
	\{(A_1,\alpha_1)\,;\,(A_2,\alpha_2)\,;\,\ldots\,;\,(A_n,\alpha_n)\}.
	$$

	Les réels $\alpha_i$ s'appellent les \textbf{coefficients} (ou \textbf{poids}) du système.
\end{definition}

\begin{attention}
	Le barycentre n'existe que si la somme des coefficients est \textbf{non nulle}.
	Si $\alpha_1 + \alpha_2 + \cdots + \alpha_n = 0$, le système n'admet pas de barycentre.
\end{attention}

\begin{propriete}{Caractérisation vectorielle}{bary_carac}
	$G$ est le barycentre de $\{(A_i, \alpha_i)\}$ si et seulement si pour tout point $M$
	du plan :
	$$
	\sum_{i=1}^{n} \alpha_i\,\vect{MA_i} = \left(\sum_{i=1}^{n} \alpha_i\right)\vect{MG}.
	$$
\end{propriete}

\begin{propriete}{Cas de deux points}{bary_deux_points}
	Le barycentre $G$ de $\{(A,\alpha)\,;\,(B,\beta)\}$ avec $\alpha + \beta \neq 0$
	vérifie :
	$$
	\vect{AG} = \frac{\beta}{\alpha+\beta}\,\vect{AB}.
	$$
	\begin{itemize}[label=\textbullet]
		\item Si $\alpha = \beta$ : $G$ est le \textbf{milieu} de $[AB]$.
		\item Si $\alpha, \beta > 0$ : $G$ est \textbf{entre} $A$ et $B$, plus près
		du point de plus grand coefficient.
		\item Si $\alpha$ et $\beta$ sont de signes opposés : $G$ est \textbf{à
		l'extérieur} de $[AB]$, du côté du point de plus grand coefficient en
		valeur absolue.
	\end{itemize}
\end{propriete}

	Soit $G$ le barycentre de $\{(A,3)\,;\,(B,1)\}$. Alors
	$\vect{AG} = \dfrac{1}{4}\vect{AB}$ : $G$ est au quart de $[AB]$ à partir de $A$.
\end{exemple}

\section{Propriétés fondamentales}
% ============================================================

\begin{propriete}{Homogénéité}{bary_homogeneite}
	Si l'on multiplie \textbf{tous} les coefficients par un même réel $\lambda \neq 0$,
	le barycentre est \textbf{inchangé} :
	$$
	\text{bar}\{(A_i,\alpha_i)\} = \text{bar}\{(A_i,\lambda\alpha_i)\}.
	$$
	On peut donc toujours se ramener à des coefficients entiers, ou de somme $1$.
\end{propriete}

\begin{propriete}{Associativité du barycentre}{bary_associativite}
	Soient $G_1$ le barycentre de $\{(A_i, \alpha_i)_{i \in I}\}$ (de somme
	$\sigma_1 \neq 0$) et $G_2$ le barycentre de $\{(A_j, \alpha_j)_{j \in J}\}$
	(de somme $\sigma_2 \neq 0$), avec $I \cap J = \emptyset$.

	Alors le barycentre du système complet $\{(A_i, \alpha_i)_{i \in I \cup J}\}$
	est le barycentre de
	$$
	\left\{(G_1,\ \sigma_1)\,;\,(G_2,\ \sigma_2)\right\}.
	$$
\end{propriete}

\remarque
Cette propriété permet de calculer un barycentre \textbf{en plusieurs étapes},
en regroupant d'abord certains points entre eux.

\begin{propriete}{Réduction vectorielle}{bary_reduction_vect}
	Pour tout point $M$ et tout système $\{(A_i, \alpha_i)\}$ :
	\begin{itemize}[label=\textbullet]
		\item si $\sigma = \sum \alpha_i \neq 0$, de barycentre $G$ :
		$\displaystyle\sum_{i=1}^{n} \alpha_i\,\vect{MA_i} = \sigma\,\vect{MG}$ ;
		\item si $\sigma = \sum \alpha_i = 0$ : le vecteur
		$\displaystyle\sum_{i=1}^{n} \alpha_i\,\vect{MA_i}$ est \textbf{constant}
		(indépendant de $M$).
	\end{itemize}
\end{propriete}

\begin{propriete}{Coordonnées du barycentre}{bary_coord}
	Dans un repère, si $A_i$ a pour coordonnées $(x_i, y_i)$, alors le barycentre
	$G$ de $\{(A_i, \alpha_i)\}$ a pour coordonnées :
	$$
	x_G = \frac{\sum \alpha_i x_i}{\sum \alpha_i},
	\qquad
	y_G = \frac{\sum \alpha_i y_i}{\sum \alpha_i}.
	$$
\end{propriete}

\begin{exemple}
	Soient $A(1\,;\,2)$, $B(3\,;\,0)$, $C(-1\,;\,4)$ avec coefficients $2$, $1$, $3$.
	$$
	x_G = \frac{2(1) + 1(3) + 3(-1)}{2+1+3} = \frac{2}{6} = \frac{1}{3},
	\qquad
	y_G = \frac{2(2) + 1(0) + 3(4)}{6} = \frac{16}{6} = \frac{8}{3}.
	$$
	Donc $G\!\left(\dfrac{1}{3}\,;\,\dfrac{8}{3}\right)$.
\end{exemple}

\begin{propriete}{Conservation par isométrie}{bary_isometrie}
	Toute isométrie (et plus généralement toute application affine) conserve le
	barycentre : l'image du barycentre de $\{(A_i,\alpha_i)\}$ est le barycentre
	de $\{(A_i',\alpha_i)\}$, où $A_i'$ est l'image de $A_i$.
\end{propriete}

\section{Centres remarquables d'un triangle}
% ============================================================

\begin{propriete}{Centres remarquables comme barycentres}{bary_centres}
	Soit $ABC$ un triangle.
	\begin{enumerate}
		\item Le \textbf{centre de gravité} $G$ est le barycentre de
		$\{(A,1)\,;\,(B,1)\,;\,(C,1)\}$. Il vérifie $\vect{GA} + \vect{GB} + \vect{GC} = \vect{0}$.

		\item L'\textbf{orthocentre} $H$ est le barycentre de
		$$\{(A,\tan\hat{A})\,;\,(B,\tan\hat{B})\,;\,(C,\tan\hat{C})\}.$$

		\item Le \textbf{centre du cercle circonscrit} $O$ est le barycentre de
		$$\{(A,\sin 2\hat{A})\,;\,(B,\sin 2\hat{B})\,;\,(C,\sin 2\hat{C})\}.$$
	\end{enumerate}
\end{propriete}

\begin{propriete}{Relation d'Euler}{bary_euler}
	Dans tout triangle $ABC$, en notant $G$ le centre de gravité,
	$H$ l'orthocentre et $O$ le centre du cercle circonscrit :
	$$
	\vect{OH} = 3\,\vect{OG}.
	$$
	Les points $O$, $G$, $H$ sont \textbf{alignés} (droite d'Euler) et $OG = \dfrac{1}{3}OH$.
\end{propriete}

\section{Lignes de niveau}
% ============================================================

\begin{definition}
	\textbf{Ligne de niveau.}\par
	Soit $\{(A_i, \alpha_i)\}$ un système de points pondérés. Pour tout réel $k$,
	l'\textbf{ensemble de niveau $k$} est :
	$$
	\mathscr{E}_k = \left\{ M \in \mathscr{P} \;\middle|\;
	\sum_{i=1}^{n} \alpha_i\,MA_i^2 = k \right\}.
	$$
\end{definition}

\begin{theoreme}[Réduction de la somme pondérée des carrés]
	Soit $\sigma = \sum \alpha_i$.
	\begin{itemize}[label=\textbullet]
		\item \textbf{Si $\sigma \neq 0$}, de barycentre $G$ : pour tout point $M$,
		$$
		\sum_{i=1}^{n} \alpha_i\,MA_i^2 = \sigma\,MG^2 + K,
		\qquad K = \sum_{i=1}^{n} \alpha_i\,GA_i^2 .
		$$
		\item \textbf{Si $\sigma = 0$}, en posant $\vec{u} = \sum \alpha_i\,\vect{OA_i}$
		(vecteur constant) : $\displaystyle\sum \alpha_i\,MA_i^2 = 2\,\vect{OM}\cdot\vec{u} + c$
		est une fonction \textbf{affine} des coordonnées de $M$.
	\end{itemize}
\end{theoreme}

\begin{propriete}{Nature des lignes de niveau}{bary_niveau_nature}
	\begin{enumerate}
		\item \textbf{Si $\sigma > 0$} : $\mathscr{E}_k$ est
		\begin{itemize}[label=\textbullet]
			\item un \textbf{cercle} de centre $G$ et de rayon $r = \sqrt{\dfrac{k-K}{\sigma}}$
			si $k > K$,
			\item le \textbf{point} $G$ si $k = K$, \quad l'\textbf{ensemble vide} si $k < K$.
		\end{itemize}
		\item \textbf{Si $\sigma < 0$} : mêmes conclusions en échangeant les
		inégalités ($\mathscr{E}_k$ cercle si $k < K$, vide si $k > K$).
		\item \textbf{Si $\sigma = 0$} : $\mathscr{E}_k$ est une \textbf{droite}
		(perpendiculaire à $\vec u$), ou $\emptyset$, ou le plan entier.
	\end{enumerate}
\end{propriete}

\begin{center}
\begin{tikzpicture}[>=Stealth, scale=1.0]
	\coordinate (G) at (0,0);
	\draw[blue!70!black,thick] (G) circle (1.8);
	\filldraw (G) circle (1.6pt) node[below right,font=\small] {$G$};
	\coordinate (M) at (35:1.8);
	\filldraw (M) circle (1.6pt) node[above right,font=\small] {$M$};
	\draw[gray!70] (G) -- (M) node[midway,above,font=\scriptsize] {$r$};
	\node[blue!70!black,font=\scriptsize] at (-1.5,-1.9)
	  {$\sum \alpha_i\,MA_i^2 = k \iff MG^2 = \dfrac{k-K}{\sigma}$};
\end{tikzpicture}
\end{center}

\begin{astucebac}
	Pour déterminer la nature d'un ensemble $\sum \alpha_i\,MA_i^2 = k$ :
	\begin{enumerate}
		\item Calculer $\sigma = \sum \alpha_i$.
		\item Si $\sigma \neq 0$ : trouver le barycentre $G$, calculer
		$K = \sum \alpha_i\,GA_i^2$, réduire en $\sigma\,MG^2 + K = k$, conclure
		selon le signe de $\dfrac{k-K}{\sigma}$.
		\item Si $\sigma = 0$ : la relation devient affine $\Rightarrow$
		\textbf{droite}. On développe $\sum \alpha_i\,MA_i^2$ avec un point fixe
		bien choisi.
	\end{enumerate}
\end{astucebac}

\section{Méthodes et exercices résolus}
% ============================================================

\begin{methode}
	\textbf{Construire le barycentre d'un système.}
	\begin{enumerate}
		\item Vérifier $\sigma = \sum \alpha_i \neq 0$.
		\item Regrouper les points par associativité (réduire à deux points
		partiels $G_1$, $G_2$).
		\item Placer chaque barycentre partiel avec $\vect{AG} = \dfrac{\beta}{\alpha+\beta}\vect{AB}$.
		\item Placer $G$ sur $[G_1G_2]$ avec les poids partiels.
	\end{enumerate}
\end{methode}

\exerciceresolu
\textbf{Construire un barycentre par associativité}

Soient $A$, $B$, $C$, $D$ quatre points. Construire le barycentre $G$ de
$$\{(A,3)\,;\,(B,-1)\,;\,(C,2)\,;\,(D,2)\}.$$

\solution

$\sigma = 3 - 1 + 2 + 2 = 6 \neq 0$, donc $G$ existe.

\textbf{Étape 1 :} $G_1 =$ barycentre de $\{(A,3)\,;\,(B,-1)\}$, poids total $2$.
$\vect{AG_1} = \dfrac{-1}{2}\,\vect{AB}$ : $G_1$ est à l'extérieur de $[AB]$, du côté de $A$.

\textbf{Étape 2 :} $G_2 =$ barycentre de $\{(C,2)\,;\,(D,2)\}$, poids total $4$ :
$G_2 = $ milieu de $[CD]$.

\textbf{Étape 3 :} Par associativité, $G =$ barycentre de $\{(G_1, 2)\,;\,(G_2, 4)\}$ :
$$
\vect{G_1 G} = \frac{4}{6}\,\vect{G_1 G_2} = \frac{2}{3}\,\vect{G_1 G_2}.
$$

\exerciceresolu
\textbf{Médianes et centre de gravité}

Soient $A$, $B$, $C$ un triangle et $I$, $J$, $K$ les milieux de $[BC]$, $[CA]$, $[AB]$.
Montrer que les médianes $[AI]$, $[BJ]$, $[CK]$ sont concourantes.

\solution

Soit $G$ le barycentre de $\{(A,1)\,;\,(B,1)\,;\,(C,1)\}$ : $\vect{GA} + \vect{GB} + \vect{GC} = \vect{0}$.

$I$ milieu de $[BC]$ $\Rightarrow$ $\vect{GB}+\vect{GC} = 2\,\vect{GI}$, donc
$\vect{GA} = -2\,\vect{GI}$ : $G \in (AI)$ avec $AG = 2\,GI$.

Par le même raisonnement $G \in (BJ)$ et $G \in (CK)$ : les trois médianes
se coupent en $G$.

\exerciceresolu
\textbf{Ligne de niveau — cercle}

Dans un repère orthonormé, $A(-2\,;\,1)$ et $B(4\,;\,3)$.
Déterminer l'ensemble $\mathscr{E}$ des points $M$ tels que $2\,MA^2 - MB^2 = 10$.

\solution

\textbf{Étape 1.} $\sigma = 2 - 1 = 1 \neq 0$. Barycentre $G$ de $\{(A,2)\,;\,(B,-1)\}$ :
$\vect{AG} = \dfrac{-1}{1}\,\vect{AB} = -\vect{AB}$, donc $G(-8\,;\,-1)$.

\textbf{Étape 2.} $GA^2 = 36 + 4 = 40$ et $GB^2 = 144 + 16 = 160$, donc
$K = 2(40) - 160 = -80$.

\textbf{Étape 3.} $2\,MA^2 - MB^2 = MG^2 - 80 = 10 \iff MG^2 = 90 \iff MG = 3\sqrt{10}$.

$\mathscr{E}$ est le \textbf{cercle} de centre $G(-8\,;\,-1)$ et de rayon $3\sqrt{10}$.

\exerciceresolu
\textbf{Ligne de niveau — droite (somme des coefficients nulle)}

$A$ et $B$ sont deux points tels que $AB = 4$. Déterminer l'ensemble des points
$M$ tels que $MA^2 - MB^2 = 8$.

\solution

$\sigma = 1 - 1 = 0$ : pas de barycentre, l'ensemble est une droite.

On introduit $I$ le milieu de $[AB]$ :
$$
MA^2 - MB^2 = \vect{MA}\cdot\vect{MA} - \vect{MB}\cdot\vect{MB}
= (\vect{MA}+\vect{MB})\cdot(\vect{MA}-\vect{MB}) = 2\,\vect{MI}\cdot\vect{BA}.
$$
Donc $MA^2 - MB^2 = 8 \iff 2\,\vect{MI}\cdot\vect{BA} = 8 \iff \vect{IM}\cdot\vect{BA} = -4$.

L'ensemble est la \textbf{droite perpendiculaire à $(AB)$} passant par le point
$H$ de $[AB]$ tel que $\vect{IH}\cdot\vect{BA} = -4$, soit $\overline{IH} = \dfrac{-4}{AB} = -1$
(mesure algébrique sur l'axe $(\,\vect{BA}\,)$).

	\item Justifier que $O$ est le milieu de $[AC]$ et de $[BD]$ ; autrement dit

\exercice{Lire une ligne de niveau \hfill \moyen}
$A$ et $B$ sont deux points, $I$ le milieu de $[AB]$, avec $AB = 4$. On note
$\mathscr{E}_k$ l'ensemble des points $M$ tels que $MA^2 + MB^2 = k$.

\begin{center}
\begin{tikzpicture}[>=Stealth, scale=1.0]
	\coordinate (A) at (-2,0); \coordinate (B) at (2,0); \coordinate (I) at (0,0);
	\draw[thick] (A)--(B);
	\draw[blue!70!black,thick] (I) circle (1.5);
	\filldraw (A) circle (1.5pt) node[below,font=\small] {$A$};
	\filldraw (B) circle (1.5pt) node[below,font=\small] {$B$};
	\filldraw (I) circle (1.4pt) node[below right,font=\small] {$I$};
	\filldraw (55:1.5) circle (1.5pt) node[above right,font=\small] {$M$};
\end{tikzpicture}
\end{center}

\begin{enumerate}
	\item Montrer que $MA^2 + MB^2 = 2\,MI^2 + \dfrac{AB^2}{2}$.
	\item Pour quelles valeurs de $k$ l'ensemble $\mathscr{E}_k$ est-il un
	cercle ? un point ? vide ?
	\item Décrire $\mathscr{E}_{20}$.
\end{enumerate}

\exercice{Vrai ou faux \hfill \facile}
Répondre par \emph{vrai} ou \emph{faux} en justifiant brièvement.
\begin{enumerate}
	\item Le barycentre de $\{(A,\alpha)\,;\,(B,\beta)\}$ existe pour tous réels
	$\alpha$, $\beta$.
	\item Si l'on multiplie tous les coefficients par $-3$, le barycentre est
	inchangé.
	\item Le barycentre de $\{(A,1)\,;\,(B,1)\,;\,(C,1)\}$ est le centre de
	gravité du triangle $ABC$.
	\item Si $\alpha + \beta + \gamma = 0$, alors $\alpha\vect{MA} + \beta\vect{MB}
	+ \gamma\vect{MC}$ dépend du point $M$.
	\item L'ensemble des points $M$ tels que $MA^2 - MB^2 = k$ est toujours un
	cercle.
	\item Une isométrie transforme le barycentre d'un système en le barycentre
	du système des images.
	\item Le barycentre de $\{(A,2)\,;\,(B,3)\}$ est plus proche de $B$ que de $A$.
\end{enumerate}

\subsection{Barycentre et lignes de niveau}

\exercice{Coordonnées d'un barycentre \hfill \facile}
Dans un repère orthonormé, $A(1\,;\,3)$, $B(5\,;\,-1)$, $C(-2\,;\,0)$.
\begin{enumerate}
	\item Calculer les coordonnées du barycentre $G$ de $\{(A,2)\,;\,(B,1)\,;\,(C,1)\}$.
	\item Calculer celles du barycentre $H$ de $\{(A,1)\,;\,(B,-1)\,;\,(C,1)\}$.
	\item Vérifier que $\vect{GH}$ est colinéaire à un vecteur simple.
\end{enumerate}

\exercice{Barycentre de deux points \hfill \facile}
Soient $A(1\,;\,3)$ et $B(5\,;\,-1)$.
\begin{enumerate}
	\item Déterminer le barycentre $G_1$ de $\{(A,3)\,;\,(B,1)\}$.
	\item Le barycentre $G_2$ de $\{(A,2)\,;\,(B,-2)\}$ existe-t-il ? Justifier.
	\item Déterminer le barycentre $G_3$ de $\{(A,1)\,;\,(B,4)\}$ et vérifier
	qu'il est entre $A$ et $B$.
\end{enumerate}

\exercice{Centre de gravité et alignements \hfill \moyen}
Soient $A(0\,;\,0)$, $B(6\,;\,0)$, $C(0\,;\,4)$.
\begin{enumerate}
	\item Calculer les coordonnées du centre de gravité $G$ du triangle $ABC$.
	\item Vérifier que $\vect{GA} + \vect{GB} + \vect{GC} = \vect{0}$.
	\item Déterminer le barycentre $H$ de $\{(A,4)\,;\,(B,1)\,;\,(C,-2)\}$.
	\item Les points $G$, $H$ et $C$ sont-ils alignés ?
\end{enumerate}

\exercice{Nature d'un ensemble de points \hfill \moyen}
On donne $A(1\,;\,0)$ et $B(-1\,;\,0)$. Déterminer la nature de l'ensemble des
points $M$ vérifiant :
\begin{enumerate}
	\item $MA^2 + MB^2 = 6$
	\item $MA^2 + MB^2 = 1$
	\item $3\,MA^2 + MB^2 = 12$
	\item $MA^2 - 3\,MB^2 = 4$
	\item $MA^2 - MB^2 = 2$
\end{enumerate}

\exercice{Ensemble de points et associativité \hfill \moyen}
$ABC$ est un triangle. $G$ est le barycentre de $\{(A,1)\,;\,(B,1)\,;\,(C,2)\}$.
\begin{enumerate}
	\item Construire $G$ (on introduira le milieu $K$ de $[AB]$).
	\item Montrer que pour tout point $M$ :
	$\vect{MA} + \vect{MB} + 2\,\vect{MC} = 4\,\vect{MG}$.
	\item Déterminer et construire l'ensemble des points $M$ tels que
	$$
	\left\| \vect{MA} + \vect{MB} + 2\,\vect{MC} \right\| = \left\| \vect{MA} + \vect{MB} \right\|.
	$$
\end{enumerate}

\exercice{Problème de synthèse \hfill \difficile}
Dans un repère orthonormé, $A(2\,;\,1)$, $B(-1\,;\,3)$, $C(4\,;\,-2)$.
\begin{enumerate}
	\item Déterminer le barycentre $G$ de $\{(A,1)\,;\,(B,2)\,;\,(C,3)\}$.
	\item Montrer que pour tout point $M$ :
	$MA^2 + 2\,MB^2 + 3\,MC^2 = 6\,MG^2 + K$, où $K$ est une constante à calculer.
	\item En déduire la nature et les éléments de l'ensemble $\mathscr{E}$ des
	points $M$ tels que $MA^2 + 2\,MB^2 + 3\,MC^2 = 30$.
	\item Déterminer le point $M_0$ de l'axe des abscisses appartenant à $\mathscr{E}$.
	\item Montrer que $G$ minimise $MA^2 + 2\,MB^2 + 3\,MC^2$ ; donner ce minimum.
\end{enumerate}

\exercice{Sujet type Bac — barycentre et discussion \hfill \difficile}
Dans le plan orienté, $ABC$ est un triangle rectangle isocèle direct en $A$ tel
que $AB = AC = 4\,$cm et $(\vect{AB},\vect{AC}) = \dfrac{\pi}{2}$.

\textbf{Partie A}
\begin{enumerate}
	\item Déterminer et construire le point $D$ barycentre du système
	$$\{(A,1)\,;\,(B,-1)\,;\,(C,1)\}.$$
	\hfill (0,25 pt $\times$ 2)
	\item Montrer que pour tout point $M$ du plan :
	$$
	MA^2 - MB^2 + MC^2 = MD^2 + K,
	$$
	où $K$ est une constante que l'on déterminera. \hfill (0,75 pt)
	\item Discuter, suivant les valeurs du réel $k$, la nature de l'ensemble
	$(E_k)$ des points $M$ vérifiant $MA^2 - MB^2 + MC^2 = k$. \hfill (0,75 pt)
	\item Vérifier que $A \in (E_0)$, puis construire $(E_0)$. \hfill (0,25 pt)
\end{enumerate}

\textbf{Partie B} — \textit{le plan est rapporté au repère orthonormé direct
$(A\,;\,\frac14\vect{AB},\frac14\vect{AC})$.}
\begin{enumerate}
	\item Donner les coordonnées de $A$, $B$, $C$ et $D$. \hfill (0,5 pt)
	\item Retrouver par le calcul la constante $K$ et l'équation de $(E_k)$. \hfill (0,5 pt)
	\item Déterminer les coordonnées des points d'intersection de $(E_0)$ avec
	l'axe des ordonnées $(AC)$. \hfill (0,5 pt)
\end{enumerate}

\exercice{Sujet type Bac — norme et ensemble de points \hfill \difficile}
$ABCD$ est un rectangle direct tel que $AB = 6\,$cm et $AD = 4\,$cm.

\textbf{Partie A}
\begin{enumerate}
	\item Prouver que $C$ est le barycentre du système
	$\{(A,-1)\,;\,(B,1)\,;\,(D,1)\}$. \hfill (0,5 pt)
	\item Soit $\vec{w} = -2\vect{MA} + \vect{MB} + \vect{MD}$. Montrer que
	$\vec{w}$ ne dépend pas de $M$ et l'exprimer à l'aide de $\vect{AB}$ et
	$\vect{AD}$. \hfill (0,5 pt)
	\item Déterminer et construire l'ensemble $(\Gamma)$ des points $M$ du plan
	vérifiant
	$$
	\left\| -\vect{MA} + \vect{MB} + \vect{MD} \right\|
	= \left\| -2\vect{MA} + \vect{MB} + \vect{MD} \right\|.
	$$
	\hfill (1 pt)
\end{enumerate}

\textbf{Partie B} — \textit{repère orthonormé direct $(A\,;\,\frac16\vect{AB},\frac14\vect{AD})$.}
\begin{enumerate}
	\item Donner les coordonnées de $A$, $B$, $C$, $D$. \hfill (0,5 pt)
	\item Retrouver que $(\Gamma)$ est un cercle dont on précisera le centre et
	le rayon. \hfill (0,75 pt)
	\item $(\Gamma)$ coupe-t-il le segment $[AC]$ ? Justifier. \hfill (0,5 pt)
\end{enumerate}
