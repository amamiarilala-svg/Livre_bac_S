% --------- ESPACEMENT DES FORMULES CENTREES (economie de pages) ---------
% Le livre compte plusieurs centaines de formules en display ; l'air par
% defaut (~10pt au-dessus et au-dessous) represente a lui seul plusieurs
% pages. On le reduit sans toucher a la taille des formules.
\AtBeginDocument{%
	\setlength{\abovedisplayskip}{4pt plus 2pt minus 2pt}%
	\setlength{\belowdisplayskip}{4pt plus 2pt minus 2pt}%
	\setlength{\abovedisplayshortskip}{2pt plus 1pt minus 1pt}%
	\setlength{\belowdisplayshortskip}{2pt plus 1pt minus 1pt}%
}

% --------- ESPACEMENT DES LISTES (economie de pages) ---------
% Reduit l'air autour et entre les items de toutes les listes
% (enumerate, itemize, description) sans toucher au corps de texte.
\setlist{topsep=2pt, itemsep=1pt, parsep=0pt, partopsep=0pt}


% --------- STYLE COMMUN : ARBRE PONDERE A 2 NIVEAUX ---------
% Usage :
%   \begin{tikzpicture}[arbrepondere]
%       \node {}
%       child { node {B}
%           child { node {A}     edge from parent node[above, pos=0.75] {p} }
%           child { node {non A} edge from parent node[below, pos=0.75] {p} }
%           edge from parent node[above, pos=0.65] {p}
%       }
%       child { ... };
%   \end{tikzpicture}
% Remplace le positionnement manuel (\path ... ++(x,y)) : la syntaxe
% imbriquee laisse TikZ calculer l'encombrement reel de l'arbre, ce qui
% coupe la hauteur d'un facteur ~1,7 par rapport a un espacement fixe
% choisi a la main (7,2 cm -> 4,2 cm sur l'arbre a 2 niveaux du chapitre
% Probabilites conditionnelles).
%
% IMPORTANT : toujours preciser pos=0.65 (niveau 1) / pos=0.75 (niveau 2)
% sur CHAQUE etiquette "edge from parent". Au pos par defaut (0.5,
% le milieu du segment), les deux etiquettes d'un meme noeud (celle du
% dessus et celle du dessous) tombent trop pres l'une de l'autre pres du
% noeud parent et se chevauchent -- une etiquette peut alors litteralement
% masquer l'autre sans qu'aucun message de compilation ne le signale.
% Deplacer les etiquettes vers l'extremite ou divergent les branches
% (pos plus grand) regle le probleme sans agrandir l'arbre.
\tikzset{
	arbrepondere/.style={
		>=stealth, grow=right,
		level distance=2.3cm,
		level 1/.style={sibling distance=2.4cm},
		level 2/.style={sibling distance=1.8cm},
		font=\small
	}
}

%-------------------  TABLEAU-METHODE (secable) -------------------
% Tableau « Situation -> Reflexe » a deux colonnes alignees, mais construit
% ligne par ligne : LaTeX peut donc couper entre deux lignes en fin de page.
%
% Un `tabular` en est incapable (bloc insecable) : place dans une boite
% `methode`, il empeche la boite de se couper, la fait sauter entiere a la
% page suivante et laisse un trou pouvant atteindre un tiers de page.
%
%   \begin{tabmethode}[0.45]{En-tete gauche}{En-tete droite}
%       \ligne{situation}{reflexe}
%   \end{tabmethode}
%
% L'argument optionnel est la largeur de la colonne de gauche, en fraction
% de la largeur disponible (0.40 par defaut).
\newlength{\tabmethodegauche}
\newlength{\tabmethodedroite}
\newcommand{\ligne}[2]{%
	\par\noindent
	\parbox[t]{\tabmethodegauche}{\raggedright\strut #1}%
	\hspace{0.03\linewidth}%
	\parbox[t]{\tabmethodedroite}{\raggedright\strut #2}%
	\par\vspace{2.5pt}%
}
\newenvironment{tabmethode}[3][0.40]{%
	\setlength{\tabmethodegauche}{#1\linewidth}%
	\setlength{\tabmethodedroite}{\dimexpr\linewidth-\tabmethodegauche-0.035\linewidth\relax}%
	\par\smallskip
	\noindent\rule{\linewidth}{0.4pt}\par\nobreak\vspace{2pt}
	\noindent
	\parbox[t]{\tabmethodegauche}{\raggedright\strut\textbf{#2}}%
	\hspace{0.03\linewidth}%
	\parbox[t]{\tabmethodedroite}{\raggedright\strut\textbf{#3}}%
	\par\nobreak\vspace{2pt}
	\noindent\rule{\linewidth}{0.4pt}\par\nobreak\vspace{3pt}
}{%
	\par\vspace{0.5pt}\noindent\rule{\linewidth}{0.4pt}\par\smallskip
}
